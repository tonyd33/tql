title: a `do` block with no statements is permitted
grammar: typescript
asserts: core, types, values

It evaluates to its final expression.

--- tql ---
import Data.List (map);

main root = do {
  map text $ children_of_kind :class_declaration root;
};

--- source ---
class A {}

--- values ---
["class A {}"]

--- types ---
main :: Node -> [String]

--- core ---
main = \root -> map text (children_of_kind :class_declaration root)
