title: a `do`-local `let` sees bindings from earlier statements
grammar: typescript
asserts: core, types, values

`prefix` is computed from `c`, bound by the preceding statement, and is then
used by the final expression.

--- tql ---
main root =
  do {
    c <- (children | is_kind :class_declaration) root;
    n <- c.name;
    let prefix = text n;
    m <- (.body | children | is_kind :method_definition) c;
    pure prefix c
  };

--- source ---
class A { m() {} n() {} }

--- values ---
["A", "A"]

--- core ---
main = \root -> bind c <- compose children is_kind[class_declaration] root in bind n <- field[name] c in letrec { prefix = text n } in bind m <- compose (compose field[body] children) is_kind[method_definition] c in pure prefix c

--- types ---
main : Node -> [String]
