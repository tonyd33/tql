title: an expression statement is a bind whose value is discarded
grammar: typescript
asserts: core, types, values

`guard $ ...;` needs no `_ <-`.

--- tql ---
main root = do {
  class_decl <- children_of_kind :class_declaration root;
  name <- #name class_decl;
  guard $ text name ~ r"^S";
  return $ text name;
};

--- source ---
class Service {}
class Other {}
class Store {}

--- values ---
["Service", "Store"]

--- types ---
main :: Node -> [String]

--- core ---
main = \root -> concat_map (\class_decl -> concat_map (\name -> concat_map (\_ -> return (text name)) (guard (op[~] (text name) r"^S"))) (field[name] class_decl)) (children_of_kind :class_declaration root)
