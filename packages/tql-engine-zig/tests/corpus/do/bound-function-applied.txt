title: a bound filter runs as a statement
grammar: typescript
asserts: core, types, values

`predicate` is bound to a filter and runs on the block's input as the final
statement.

--- tql ---
make_predicate = pure (arr kind);
main = do {
  predicate <- make_predicate;
  predicate;
};

--- source ---
class A {}

--- values ---
["program"]

--- types ---
make_predicate :: Kleisli List Node (Kleisli List Node Kind)
main :: Kleisli List Node Kind
entry :: Node -> [Kind]

--- core ---
make_predicate = pure (arr kind)
main = bind make_predicate (\predicate -> predicate)
entry = run_main main
