title: `guard` in a do block drops the outputs whose condition is false
grammar: typescript
asserts: types, values

--- tql ---
main root =
  do {
    c <- (children | is_kind :class_declaration) root;
    n <- .name c;
    _ <- guard (text n ~ r"^S");
    [text n]
  };

--- source ---
class Service {}
class Other {}
class Store {}

--- values ---
["Service", "Store"]
