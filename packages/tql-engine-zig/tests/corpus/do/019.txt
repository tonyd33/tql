title: `guard` in a do block drops the outputs whose condition is false
grammar: typescript
asserts: core, types, values

--- tql ---
main root = do {
  c <- children_of_kind :class_declaration root;
  n <- c#name;
  _ <- guard $ text n ~ r"^S";
  return $ text n;
};

--- source ---
class Service {}
class Other {}
class Store {}

--- values ---
["Service", "Store"]

--- types ---
main :: Node -> [String]

--- core ---
main = \root -> concat_map (\c -> concat_map (\n -> concat_map (\_ -> return (text n)) (guard (op[~] (text n) r"^S"))) (field[name] c)) (children_of_kind :class_declaration root)
