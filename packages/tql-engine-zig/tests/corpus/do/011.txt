title: a bound node is the argument to a later navigation
grammar: typescript
asserts: core, types, values

--- tql ---
main root = do {
  c <- children_of_kind :class_declaration root;
  p <- descendants_of_kind :property_identifier c;
  return $ text p;
};

--- source ---
class A { m() {} n() {} }

--- values ---
["m", "n"]

--- types ---
main :: Node -> [String]

--- core ---
main = \root ->
  bind c <- children_of_kind :class_declaration root in
  bind p <- descendants_of_kind :property_identifier c in
  return (text p)
