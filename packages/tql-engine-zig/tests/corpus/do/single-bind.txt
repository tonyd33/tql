title: a single stream bind runs the body once per output
grammar: typescript
asserts: core, types, values

--- tql ---
main = do {
  c <- children_of_kind :class_declaration;
  return $ text c;
};

--- source ---
class A {}
class B {}

--- values ---
["class A {}", "class B {}"]

--- types ---
main :: Filter Node String

--- core ---
main = bind (children_of_kind :class_declaration) (\c -> return (text c))
