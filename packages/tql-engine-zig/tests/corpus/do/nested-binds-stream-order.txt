title: nested binds concatenate in stream order
grammar: typescript
asserts: core, types, values

Both methods of A precede the method of B: the outer bind's order is preserved
and the inner bind's results are concatenated within each branch.

--- tql ---
main = do {
  c <- children_of_kind :class_declaration;
  m <- pure c | #body | children | of_kind :method_definition;
  n <- pure m | #name;
  return $ text n;
};

--- source ---
class A { m() {} n() {} }
class B { p() {} }

--- values ---
["m", "n", "p"]

--- types ---
main :: Filter Node String

--- core ---
main = bind (children_of_kind :class_declaration) (\c -> bind (compose_flipped (compose_flipped (compose_flipped (pure c) (Filter field[body])) children) (of_kind :method_definition)) (\m -> bind (compose_flipped (pure m) (Filter field[name])) (\n -> return (text n))))
