title: `do` nested inside a pipe stage
grammar: typescript
asserts: core, types, values

The block runs once per input value from the left stage.

--- tql ---
main = children | of_kind :class_declaration | do {
  m <- #body | children | of_kind :method_definition;
  n <- pure m | #name;
  return $ text n;
};

--- source ---
class A { m() {} }
class B { p() {} }

--- values ---
["m", "p"]

--- types ---
main :: Kleisli List Node String

--- core ---
main = compose_flipped (compose_flipped children (of_kind :class_declaration)) (bind (compose_flipped (compose_flipped (Kleisli field[body]) children) (of_kind :method_definition)) (\m -> bind (compose_flipped (pure m) (Kleisli field[name])) (\n -> return (text n))))
