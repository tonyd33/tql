title: a `let`-bound scalar is an ordinary operator operand
grammar: typescript
asserts: core, types, values

`expected` is a bound `String`, compared as an ordinary operand of `=`.

--- tql ---
main root = do {
  c <- children_of_kind :class_declaration root;
  let expected = "A";
  n <- c#name;
  return $ text n = expected;
};

--- source ---
class A {}
class B {}

--- values ---
[true, false]

--- types ---
main :: Node -> [Bool]

--- core ---
main = \root -> concat_map (
  \c ->
    let expected = "A" in
    concat_map (\n -> return (eq (text n) expected)) (field[name] c)
) (children_of_kind :class_declaration root)
