title: a `do`-local `let` has no `in` and scopes over the rest of the block
grammar: typescript
asserts: core, types, values

--- tql ---
main = do {
  c <- children_of_kind :class_declaration;
  n <- pure c | #name;
  let name = text n;
  return name;
};

--- source ---
class A {}
class B {}

--- values ---
["A", "B"]

--- types ---
main :: Filter Node String

--- core ---
main = bind (children_of_kind :class_declaration) (
  \c -> bind (compose_flipped (pure c) (Filter field[name])) (
    \n ->
      let name = text n in
      return name
  )
)
