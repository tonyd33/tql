title: a `do`-local `let` has no `in` and scopes over the rest of the block
grammar: typescript
asserts: core, types, values

--- tql ---
main root = do {
  c <- children_of_kind :class_declaration root;
  n <- c#name;
  let name = text n;
  return name;
};

--- source ---
class A {}
class B {}

--- values ---
["A", "B"]

--- types ---
main :: Node -> [String]

--- core ---
main = \root -> bind[List] (children_of_kind :class_declaration root) (
  \c -> bind[List] (field[name] c) (
    \n ->
      let name = text n in
      return name
  )
)
