title: a `do`-local `let` sees bindings from earlier statements
grammar: typescript
asserts: core, types, values

`prefix` is computed from `n`, bound by the preceding statement, and is then
used by the final expression after a further bind.

--- tql ---
main = do {
  c <- children_of_kind :class_declaration;
  n <- pure c | #name;
  let prefix = text n;
  m <- pure c | #body | children | of_kind :method_definition;
  return prefix;
};

--- source ---
class A { m() {} n() {} }

--- values ---
["A", "A"]

--- core ---
main = bind (children_of_kind :class_declaration) (
  \c -> bind (compose_flipped (pure c) (Filter field[name])) (
    \n ->
      let prefix = text n in
      bind (compose_flipped (compose_flipped (compose_flipped (pure c) (Filter field[body])) children) (of_kind :method_definition)) (\m -> return prefix)
  )
)

--- types ---
main :: Filter Node String
