title: a bound filter is applied like any function
grammar: typescript
asserts: core, types, values

`predicate` is bound to a filter and applied to `root` directly.

--- tql ---
make_predicate = pure (arr kind);
main root = do {
  predicate <- make_predicate root;
  predicate root;
};

--- source ---
class A {}

--- values ---
["program"]

--- types ---
make_predicate :: a -> [Node -> [Kind]]
main :: Node -> [Kind]

--- core ---
make_predicate = pure (arr kind)
main = \root ->
  bind predicate <- make_predicate root in
  predicate root
