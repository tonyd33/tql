title: the final expression of a do block needs no trailing `;`
grammar: typescript
asserts: core, types, values

--- tql ---
main = do {
  c <- children_of_kind :class_declaration;
  m <- pure c | #body | children | of_kind :method_definition;
  n <- pure m | #name;
  return $ text n
};

--- source ---
class A { m() {} }

--- values ---
["m"]

--- types ---
main :: Kleisli List Node String

--- core ---
main = bind (children_of_kind :class_declaration) (\c -> bind (compose_flipped (compose_flipped (compose_flipped (pure c) (Kleisli field[body])) children) (of_kind :method_definition)) (\m -> bind (compose_flipped (pure m) (Kleisli field[name])) (\n -> return (text n))))
