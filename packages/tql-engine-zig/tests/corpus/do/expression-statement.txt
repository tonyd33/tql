title: an expression statement is a bind whose value is discarded
grammar: typescript
asserts: core, types, values

`guard $ ...;` needs no `_ <-`; the explicit `_ <-` spelling binds the same
way.

--- tql ---
main root = do {
  class_decl <- children_of_kind :class_declaration root;
  name <- #name class_decl;
  guard $ text name ~ r"^S";
  _ <- guard $ text name ~ r"e$";
  return $ text name;
};

--- source ---
class Service {}
class Other {}
class Store {}

--- values ---
["Service", "Store"]

--- types ---
main :: Node -> [String]

--- core ---
main = \root -> bind (children_of_kind :class_declaration root) (\class_decl -> bind (field[name] class_decl) (\name -> bind (guard (op[~] (text name) r"^S")) (\_ -> bind (guard (op[~] (text name) r"e$")) (\_ -> return (text name)))))
