title: an expression statement is a bind whose value is discarded
grammar: typescript
asserts: core, types, values

`guard $ ...;` needs no `_ <-`; the explicit `_ <-` spelling binds the same
way.

--- tql ---
main = do {
  class_decl <- children_of_kind :class_declaration;
  name <- pure class_decl | #name;
  guard $ text name ~ r"^S";
  _ <- guard $ text name ~ r"e$";
  return $ text name;
};

--- source ---
class Service {}
class Other {}
class Store {}

--- values ---
["Service", "Store"]

--- types ---
main :: Filter Node String

--- core ---
main = bind (children_of_kind :class_declaration) (\class_decl -> bind (compose_flipped (pure class_decl) (Filter field[name])) (\name -> bind (guard (op[~] (text name) r"^S")) (\_ -> bind (guard (op[~] (text name) r"e$")) (\_ -> return (text name)))))
