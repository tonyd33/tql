title: the block's argument stays in scope after later binds
grammar: typescript
asserts: core, types, values

`root` is `main`'s parameter, so the final expression can still read the
program node after two binds.

--- tql ---
main root = do {
  c <- children_of_kind :class_declaration root;
  cn <- c#name;
  return { root = kind root, class = text cn };
};

--- source ---
class A {}

--- values ---
[{ "class": "A", "root": "program" }]

--- types ---
main :: Node -> [{class: String, root: Kind}]

--- core ---
main = \root -> concat_map (\c -> concat_map (\cn -> return (record[class,root] (text cn) (kind root))) (field[name] c)) (children_of_kind :class_declaration root)
