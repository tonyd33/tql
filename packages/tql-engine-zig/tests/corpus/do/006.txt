title: a `do`-local `let` has no `in` and scopes over the rest of the block
grammar: typescript
asserts: types, values

--- tql ---
main root =
  do {
    c <- (children | is_kind :class_declaration) root;
    n <- c.name;
    let name = text n;
    pure name root
  };

class A {}
