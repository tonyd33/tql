title: a `do`-local `let` group binds several names
grammar: typescript
asserts: core, types, values

Two bindings in one group, both used in the result record.

--- tql ---
main = do {
  c <- children_of_kind :class_declaration;
  cn <- pure c | #name;
  let {
    name = text cn;
    k = kind c
  };
  return { n = name, k = k };
};

--- source ---
class A {}

--- values ---
[{ "k": "class_declaration", "n": "A" }]

--- types ---
main :: Filter Node {k: Kind, n: String}

--- core ---
main = bind (children_of_kind :class_declaration) (
  \c -> bind (compose_flipped (pure c) (Filter field[name])) (
    \cn ->
      let name = text cn in
      let k = kind c in
      return (record[k,n] k name)
  )
)
