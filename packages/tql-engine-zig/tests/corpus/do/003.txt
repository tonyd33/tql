title: nested binds concatenate in stream order
grammar: typescript
asserts: core, types, values

Both methods of A precede the method of B: the outer bind's order is preserved
and the inner bind's results are concatenated within each branch.

--- tql ---
main root = do {
  c <- children_of_kind :class_declaration root;
  m <- (#body | children | of_kind :method_definition) c;
  n <- m#name;
  return $ text n;
};

--- source ---
class A { m() {} n() {} }
class B { p() {} }

--- values ---
["m", "n", "p"]

--- types ---
main :: Node -> [String]

--- core ---
main = \root ->
  bind c <- children_of_kind :class_declaration root in
  bind m <- kleisli (kleisli field[body] children) (of_kind :method_definition) c in
  bind n <- field[name] m in
  return (text n)
