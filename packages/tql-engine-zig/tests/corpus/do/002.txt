title: a bound name refers to the bound value, not the block's argument
grammar: typescript
asserts: core, types, values

The final expression reports the kind of `c`, the class node, rather than the
root.

--- tql ---
main root = do {
  c <- children_of_kind :class_declaration root;
  return $ kind c;
};

--- source ---
class A {}

--- values ---
["class_declaration"]

--- types ---
main :: Node -> [Kind]

--- core ---
main = \root ->
  bind c <- children_of_kind :class_declaration root in
  return (kind c)
