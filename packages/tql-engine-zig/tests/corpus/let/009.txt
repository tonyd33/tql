`is_even` and `is_odd` refer to each other. Both are checked monomorphically as
one component, then generalized together.

--- tql ---
main =
  let {
    is_even n = if n = 0 then true else is_odd (n - 1);
    is_odd n = if n = 0 then false else is_even (n - 1)
  } in
    pure (is_even 4), pure (is_even 7);
