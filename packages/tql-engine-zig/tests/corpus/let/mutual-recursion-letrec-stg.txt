title: mutual recursion inside one group
grammar: typescript
asserts: core, types, values, stg

`is_even` and `is_odd` refer to each other. Both are checked monomorphically as
one component, then generalized together.

--- tql ---
main =
  let {
    is_even n = if n == 0 then True else is_odd (n - 1);
    is_odd n = if n == 0 then False else is_even (n - 1);
  }
  in Kleisli $ \_ -> [is_even 4, is_even 7];

main =
  letrec
    is_even = \n -> case eq n 0 of
      False -> is_odd (op[-] n 1)
      True -> True
    is_odd = \n -> case eq n 0 of
      False -> is_even (op[-] n 1)
      True -> False
  in
  Kleisli (\_ -> Cons (is_even 4) (Cons (is_even 7) Nil))
entry = run_main main
