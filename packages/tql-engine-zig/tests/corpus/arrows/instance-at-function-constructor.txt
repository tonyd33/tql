title: an instance head may be `(->)` short of arguments
grammar: typescript
asserts: types, values

--- tql ---
class Mappable f where {
  mapf :: (a -> b) -> f a -> f b;
};

instance Mappable ((->) r) where {
  mapf f g x = f (g x);
};

main = pure $ mapf (+ 1) (\x -> x * 2) 3;
