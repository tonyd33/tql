title: a jump's computed argument is bound above the join point's depth
grammar: typescript
asserts: core, simplified, stg, values

The view's value is the jump's argument. Its thunk is bound after the join
point, and the jump reads it before dropping back to the join point's depth.

--- tql ---
import Data.Function (const);

total xs = case xs of { [] -> 0; h : t -> h + total t };
f xs = case xs of { [x] -> x; (total -> t) -> t + t * 2 + t * 3 + t * 4 + t * 5 };
main = const [f [7], f [1, 2], f []];

--- simplified ---
total = \xs -> case xs of
  Nil -> 0
  Cons h t -> op[+] h (total t)
f = \xs ->
  join alternative t = op[+] (op[+] (op[+] (op[+] t (op[*] t 2)) (op[*] t 3)) (op[*] t 4)) (op[*] t 5) in
  case xs of
    Nil -> jump alternative (total xs)
    Cons x cons1 -> case cons1 of
      Nil -> x
      Cons _ _ -> jump alternative (total xs)
main = const (Cons (f (Cons 7 Nil)) (Cons (f (Cons 1 (Cons 2 Nil))) (Cons (f Nil) Nil)))
