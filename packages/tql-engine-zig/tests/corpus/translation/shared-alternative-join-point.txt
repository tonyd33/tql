title: a shared alternative too large to copy is a join point
grammar: typescript
asserts: core, simplified, stg, values

`alternative` is bound in `size`'s frame with no closure, and each leaf
jumps to it.

--- tql ---
import Data.Function (const);

total xs = case xs of { [] -> 0; h : t -> h + total t };
size xs = case xs of { [x] -> 1; ys -> total ys + total ys * 2 + total ys * 3 };
main = const [size [1], size [1, 2]];

--- simplified ---
total = \xs -> case xs of
  Nil -> 0
  Cons h t -> op[+] h (total t)
size = \xs ->
  join alternative ys = op[+] (op[+] (total ys) (op[*] (total ys) 2)) (op[*] (total ys) 3) in
  case xs of
    Nil -> jump alternative xs
    Cons x cons1 -> case cons1 of
      Nil -> 1
      Cons _ _ -> jump alternative xs
main = const (Cons (size (Cons 1 Nil)) (Cons (size (Cons 1 (Cons 2 Nil))) Nil))
