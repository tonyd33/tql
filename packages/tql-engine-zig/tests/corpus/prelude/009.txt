Both helpers are applied to the same source and predicate, confirming they
accept identical argument types.

--- tql ---
main root =
  do {
    c <- (children | is_kind :class_declaration) root;
    { some = any (.body | children | is_kind :method_definition)
                 (.name | lift (\n -> text n ~ r"^get")),
      every = all (.body | children | is_kind :method_definition)
                  (.name | lift (\n -> text n ~ r"^get")) } c
  };
