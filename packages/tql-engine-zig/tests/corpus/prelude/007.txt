title: `all` over a filter's outputs is true only when every output satisfies it
grammar: typescript
asserts: types, values

--- tql ---
method_names = .body | children | is_kind :method_definition | .name;
is_getter n = text n ~ r"^get";

main root = do {
  c <- children_of_kind :class_declaration root;
  return $ all is_getter (method_names c);
};

--- source ---
class A { getX() {} getY() {} }
class B { getZ() {} setW() {} }

--- values ---
[true, false]

--- types ---
method_names : Node -> [Node]
is_getter : Node -> Bool
main : Node -> [Bool]
