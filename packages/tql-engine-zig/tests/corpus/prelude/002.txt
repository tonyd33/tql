title: `guard $ has q x` drops `x` when `q x` yields nothing
grammar: typescript
asserts: types, values

--- tql ---
getters c = do {
  m <- (.body | children | of_kind :method_definition) c;
  n <- m.name;
  guard $ text n ~ r"^get";
  return n;
};

main root = do {
  c <- children_of_kind :class_declaration root;
  guard $ has getters c;
  n <- c.name;
  return $ text n;
};
