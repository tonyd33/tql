title: `has` over an empty source is false
grammar: typescript
asserts: types, values

--- tql ---
getters c = do {
  m <- (.body | children | is_kind :method_definition) c;
  n <- m.name;
  guard $ text n ~ r"^get";
  return n;
};

main root = do {
  c <- children_of_kind :class_declaration root;
  return $ has getters c;
};

--- source ---
class B {}

--- values ---
[false]

--- types ---
getters : Node -> [Node]
main : Node -> [Bool]
