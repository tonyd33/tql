title: `keep` passes its input through when a plain predicate holds
grammar: typescript
asserts: types, values

--- tql ---
starts_with_s n = text n ~ r"^S";
named_s c = any starts_with_s (#name c);

main =
  children | of_kind :class_declaration
  | keep named_s
  | #name | arr text;

class Service {}
class Other {}
class Store {}

--- values ---
["Service", "Store"]
