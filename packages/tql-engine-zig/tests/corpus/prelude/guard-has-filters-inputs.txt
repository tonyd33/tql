title: `guard $ has q x` keeps `x` when `q x` yields something
grammar: typescript
asserts: types, values

--- tql ---
getters = do {
  m <- #body | children | of_kind :method_definition;
  n <- pure m | #name;
  guard $ text n ~ r"^get";
  return n;
};

main = do {
  c <- children_of_kind :class_declaration;
  guard $ has getters c;
  n <- pure c | #name;
  return $ text n;
};

--- source ---
class Service { getX() {} }
class Other { setY() {} }
class Store { getA() {} }

--- values ---
["Service", "Store"]

--- types ---
getters :: Filter Node Node
main :: Filter Node String
