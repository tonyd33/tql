title: `all` over an empty source is true
grammar: typescript
asserts: types, values

--- tql ---
method_names = #body | children | of_kind :method_definition | #name;
is_getter n = text n ~ r"^get";

main root = do {
  c <- children_of_kind :class_declaration root;
  return $ all is_getter (method_names c);
};

--- source ---
class B {}

--- values ---
[true]

--- types ---
method_names :: Node -> [Node]
is_getter :: Node -> Bool
main :: Node -> [Bool]
