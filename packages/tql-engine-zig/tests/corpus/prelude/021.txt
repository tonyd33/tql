title: `guard $ not (has q x)` keeps `x` when `q x` yields nothing
grammar: typescript
asserts: types, values

--- tql ---
runs c = do {
  m <- (.body | children | is_kind :method_definition) c;
  n <- m.name;
  guard $ text n = "run";
  return n;
};

main root = do {
  c <- children_of_kind :class_declaration root;
  guard $ not (has runs c);
  n <- c.name;
  return $ text n;
};

--- source ---
class A { run() {} }
class B { stop() {} }
class C {}

--- values ---
["B", "C"]

--- types ---
runs : Node -> [Node]
main : Node -> [String]
