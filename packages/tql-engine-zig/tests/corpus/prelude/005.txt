title: `has` is true when some output of the query passes its guard
grammar: typescript
asserts: types, values

--- tql ---
getters c = do {
  m <- (.body | children | of_kind :method_definition) c;
  n <- m.name;
  guard $ text n ~ r"^get";
  return n;
};

main root = do {
  c <- children_of_kind :class_declaration root;
  return $ has getters c;
};

--- source ---
class A { getX() {} setY() {} }
class B { setZ() {} }

--- values ---
[true, false]

--- types ---
getters : Node -> [Node]
main : Node -> [Bool]
