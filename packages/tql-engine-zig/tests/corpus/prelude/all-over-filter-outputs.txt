title: `all` over a filter's outputs is true only when every output satisfies it, and true when there are none
grammar: typescript
asserts: types, values

--- tql ---
import Data.Foldable (all);

method_names = #body | children | of_kind :method_definition | #name;
is_getter n = text n ~ r"^get";

main = do {
  c <- children_of_kind :class_declaration;
  return $ all is_getter (run method_names c);
};

--- source ---
class A { getX() {} getY() {} }
class B { getZ() {} setW() {} }
class C {}

--- values ---
[true, false, true]

--- types ---
method_names :: Filter Node Node
is_getter :: Node -> Bool
main :: Filter Node Bool
