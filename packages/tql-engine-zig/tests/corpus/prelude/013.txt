title: `has` over `ancestors` is false when no ancestor matches
grammar: typescript
asserts: types, values

--- tql ---
main root = do {
  n <- descendants_of_kind :property_identifier root;
  return $ has (ancestors | of_kind :interface_declaration) n;
};

--- source ---
class A { m() {} }

--- values ---
[false]
