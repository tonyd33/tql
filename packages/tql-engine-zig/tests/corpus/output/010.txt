title: nested lists and records encode recursively
grammar: typescript
asserts: types, values

--- tql ---
main root =
  [\r -> do {
    c <- (children | is_kind :class_declaration) r;
    n <- c.name;
    methods <- [.body | children | is_kind :method_definition | .name | lift text] c;
    pure { name = text n, methods = methods } r
  }] root;

--- source ---
class A { m() {} }
class B {}

--- values ---
[
  [
    { "methods": ["m"], "name": "A" },
    { "methods": [], "name": "B" }
  ]
]

--- types ---
main : Node -> [[{methods: [String], name: String}]]
