title: `filename` composes with other projections
grammar: typescript
asserts: types, values
file: src/service.ts

`filename` runs on the node input like any other filter, so it can appear as a
record field beside node projections.

--- tql ---
main root =
  do {
    c <- (children | is_kind :class_declaration) root;
    n <- c.name;
    f <- filename c;
    pure { file = f, name = text n } root
  };

--- source ---
class A {}

--- values ---
[{ "file": "src/service.ts", "name": "A" }]

--- types ---
main : Node -> [{file: String, name: String}]
