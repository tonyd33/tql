title: navigation binds tighter than application
grammar: typescript
asserts: core, types, values

`map text n#name` groups as `map text (n#name)`, not `(map text n)#name`.
`n#name` is a `[Node]`, and `map text` maps over it.

--- tql ---

name_of = Kleisli $ \n -> map text n#name;
main = children | of_kind :class_declaration | name_of;

--- source ---
class A {}

--- values ---
["A"]

--- types ---
name_of :: Kleisli List Node String
main :: Kleisli List Node String
entry :: Node -> [Json]

--- core ---
name_of = Kleisli (\n -> map text (field[name] n))
main = compose_flipped (compose_flipped children (of_kind :class_declaration)) name_of
entry = run_main main
