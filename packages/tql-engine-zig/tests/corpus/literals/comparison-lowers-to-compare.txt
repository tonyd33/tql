main = Filter (
  \_ -> Cons (
    case compare 1 2 of
      LT -> True
      EQ -> False
      GT -> False
  ) (
    Cons (
      case compare 2 2 of
        LT -> True
        EQ -> True
        GT -> False
    ) (
      Cons (
        case compare 3 4 of
          LT -> False
          EQ -> False
          GT -> True
      ) (

          case compare 3 3 of
            LT -> False
            EQ -> True
            GT -> True
        ) Nil
      )
    )
  )
