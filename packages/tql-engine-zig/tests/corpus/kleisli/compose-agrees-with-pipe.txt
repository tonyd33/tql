import Control.Arrow (Kleisli(..), run_kleisli);

piped :: Filter Node String;
piped = named_children | named_children | arr kind_name;

composed :: Node -> [String];
composed = run_kleisli (compose (arr kind_name) (compose (Kleisli (run named_children)) (Kleisli (run named_children))));

dotted :: Node -> [String];
dotted = run_kleisli (arr kind_name . Kleisli (run named_children) . Kleisli (run named_children));

main = Filter (\root -> [run piped root = composed root, run piped root = dotted root]);
