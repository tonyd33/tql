title: `do` over `Kleisli List Node` gives every statement the same node
grammar: typescript
asserts: types, values, simplified

--- tql ---
import Control.Arrow (Kleisli(..), run_kleisli);
import Control.Category (Category(..));

ask :: Kleisli List Node Node;
ask = identity;

pairs :: Kleisli List Node (String, String);
pairs = do {
  n <- ask;
  c <- Kleisli named_children;
  return (kind_name n, kind_name c);
};

main = descendants_of_kind :class_declaration | run_kleisli pairs;

--- source ---
class A { b() {} }

--- values ---
[["class_declaration","type_identifier"],["class_declaration","class_body"]]

--- simplified ---
pairs = \x -> bind[List] (Cons x Nil) (\y -> bind[List] (%named_children x) (\y' -> Cons ((,) (%kind_name y) (%kind_name y')) Nil))
main =
  let p = %descendants_of_kind :class_declaration in
  \x -> bind[List] (p x) pairs

--- types ---
ask :: Kleisli List Node Node
pairs :: Kleisli List Node (String, String)
main :: Node -> [(String, String)]
