title: an alias may name another alias, declared before or after it
grammar: typescript
asserts: types, values

--- tql ---
type Ints = Both Int;
type Both a = {first: a, second: a};
total :: Ints -> Int;
total b = b.first + b.second;
main = pure $ total {first = 1, second = 2};
