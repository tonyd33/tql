title: an alias of a function type
grammar: typescript

--- tql ---
type Pred a = a -> Bool;
both :: Pred a -> Pred a -> Pred a;
both p q x = p x and q x;
main = pure $ both (\n -> n > 1) (\n -> n < 5) 3;
