--- simplified ---
f = \b -> \x ->
  let a = op[*] x x in
  case b of
    False -> op[*] a 2
    True -> op[+] a 1
main = const (Cons (op[+] (op[*] 3 3) 1) (Cons (op[*] (op[*] 3 3) 2) Nil))
