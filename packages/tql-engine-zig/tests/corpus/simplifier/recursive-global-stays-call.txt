title: a recursive global function stays a call
grammar: typescript
asserts: core, simplified, values

--- tql ---
count n = if n == 0 then 0 else 1 + count (n - 1);
main = pure (count 3);

--- source ---
class A {}

--- values ---
[3]

--- simplified ---
count = \n -> case %eq_int n 0 of
  False -> op[+] 1 (count (op[-] n 1))
  True -> 0
main =
  let y = count 3 in
  \_ -> Cons y Nil

--- core ---
count = \n -> case eq n 0 of
  False -> op[+] 1 (count (op[-] n 1))
  True -> 0
main = pure (count 3)
