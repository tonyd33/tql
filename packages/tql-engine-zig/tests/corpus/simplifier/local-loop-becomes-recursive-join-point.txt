title: a local loop called only in tail position becomes a recursive join point
grammar: typescript
asserts: simplified, stg, values

The `case` around the loop moves into its body, so the loop answers `"yes"`
or `"no"` directly.

--- tql ---
import Data.Function (const);

f xs = case (
  let {
    go ys = case ys of {
      [] -> False;
      h : t -> if h = 3 then True else go t;
    };
  } in go xs
) of {
  True -> "yes";
  False -> "no";
};
main = const [f [1, 3], f [4]];

--- source ---
class A {}

--- values ---
["yes","no"]

--- simplified ---
f = \xs ->
  joinrec
    go ys = case ys of
      Nil -> "no"
      Cons h t -> case %eq_int h 3 of
        False -> jump go t
        True -> "yes"
  in
  jump go xs
main = const (Cons (f (Cons 1 (Cons 3 Nil))) (Cons (f (Cons 4 Nil)) Nil))
