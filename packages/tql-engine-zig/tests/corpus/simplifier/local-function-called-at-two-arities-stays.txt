title: a local function called with two numbers of arguments stays a function
grammar: typescript
asserts: simplified, values

--- tql ---
f b n =
  let { g x = seq (n * 2 + n * 3 + n * 4 + n * 5) x }
  in if b then g (\y -> y + 1) 3 else g 3;
