title: a case of an inlined helper's case takes the known constructor and fuses the kind test
grammar: typescript
asserts: core, simplified, stg, values

`only` inlines to a `case` on `is_kind`. Case of case moves the outer
alternatives into it, each inner alternative builds a constructor the outer
`case` knows, and the remaining kind test fuses into the axis.

--- tql ---
only k x = case is_kind k x of { False -> []; True -> [x] };
main root = do {
  n <- descendants root;
  case only :identifier n of { [] -> []; h : _ -> [text h] };
};

--- source ---
let a = b;

--- values ---
["a","b"]
--- simplified ---
only = \k -> \x -> case %is_kind k x of
  False -> Nil
  True -> Cons x Nil
main = \root -> concat_map (\n -> Cons (%text n) Nil) (%descendants_of_kind :identifier root)

--- core ---
only = \k -> \x -> case is_kind k x of
  False -> Nil
  True -> Cons x Nil
main = \root -> concat_map (
  \n -> case only :identifier n of
    Nil -> Nil
    Cons h _ -> Cons (text h) Nil
) (descendants root)
