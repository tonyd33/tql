title: a let-bound scrutinee used in an alternative becomes the case binder
grammar: typescript
asserts: simplified, stg, values

--- tql ---
double :: [Int] -> [Int];
double xs = case xs of {
  [] -> [];
  h : t -> h * 2 : double t;
};
f :: [Int] -> Int;
f xs = let { ys = double xs; } in case ys of {
  [] -> 0;
  h : _ -> h + length ys;
};
main = pure [f [], f [1, 2, 3]];
