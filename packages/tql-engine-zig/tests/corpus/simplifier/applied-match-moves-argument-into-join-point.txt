title: a match applied to an argument moves it into the shared alternative
grammar: typescript
asserts: core, simplified, values

The argument moves into every alternative, and into `alternative`'s value.
Each jump drops it, since `alternative` already took it.

--- tql ---
import Data.Function (const);

total xs = case xs of { [] -> 0; h : t -> h + total t };
pick xs = (case xs of { [x] -> \y -> y + x; ys -> \y -> y * total ys + total ys * 2 + total ys * 3 }) 10;
main = const [pick [1], pick [1, 2], pick []];

--- source ---
class A {}

--- simplified ---
total = \xs -> case xs of
  Nil -> 0
  Cons h t -> op[+] h (total t)
pick = \xs ->
  join alternative ys = op[+] (op[+] (op[*] 10 (total ys)) (op[*] (total ys) 2)) (op[*] (total ys) 3) in
  case xs of
    Nil -> jump alternative xs
    Cons x cons1 -> case cons1 of
      Nil -> op[+] 10 x
      Cons _ _ -> jump alternative xs
main = const (Cons (pick (Cons 1 Nil)) (Cons (pick (Cons 1 (Cons 2 Nil))) (Cons (pick Nil) Nil)))
