title: a let in a case scrutinee floats out of the case
grammar: typescript
asserts: core, simplified, values

--- tql ---
import Data.Function (const);

total xs = case xs of { [] -> 0; h : t -> h + total t };
f xs = case (let { s = total xs; } in s * s > 10) of { True -> "big"; False -> "small" };
main = const [f [1, 2], f [5]];

--- source ---
class A {}

--- values ---
["small","big"]
--- simplified ---
total = \xs -> case xs of
  Nil -> 0
  Cons h t -> op[+] h (total t)
f = \xs ->
  let s = total xs in
  case %gt_int (op[*] s s) 10 of
    False -> "small"
    True -> "big"
main = const (Cons (f (Cons 1 (Cons 2 Nil))) (Cons (f (Cons 5 Nil)) Nil))
--- core ---
total = \xs -> case xs of
  Nil -> 0
  Cons h t -> op[+] h (total t)
f = \xs -> case (
  let s = total xs in
  case compare (op[*] s s) 10 of
    LT -> False
    EQ -> False
    GT -> True
) of
  False -> "small"
  True -> "big"
main = const (Cons (f (Cons 1 (Cons 2 Nil))) (Cons (f (Cons 5 Nil)) Nil))
