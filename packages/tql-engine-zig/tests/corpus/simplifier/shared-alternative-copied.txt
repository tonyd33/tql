--- core ---
size = \xs ->
  join alternative ys = length ys in
  case xs of
    Nil -> jump alternative xs
    Cons x cons1 -> case cons1 of
      Nil -> 1
      Cons _ _ -> jump alternative xs
main = const (map size (Cons (Cons 1 Nil) (Cons (Cons 1 (Cons 2 Nil)) Nil)))
