import Data.Function (const);

size xs = case xs of { [x] -> 1; ys -> length ys };
main = const [size [1], size [1, 2]];

class A {}

--- simplified ---
size = \xs -> case xs of
  Nil -> length[List] xs
  Cons x cons1 -> case cons1 of
    Nil -> 1
    Cons _ _ -> length[List] xs
main =
  let
    x = Cons 1 (
      Cons (
        let xs = Cons 1 (Cons 2 Nil) in
        case xs of
          Nil -> length[List] xs
          Cons x cons1 -> case cons1 of
            Nil -> 1
            Cons _ _ -> length[List] xs
      ) Nil
    )
  in
  \y -> x

--- core ---
size = \xs ->
  join alternative ys = length ys in
  case xs of
    Nil -> jump alternative xs
    Cons x cons1 -> case cons1 of
      Nil -> 1
      Cons _ _ -> jump alternative xs
main = const (Cons (size (Cons 1 Nil)) (Cons (size (Cons 1 (Cons 2 Nil))) Nil))
