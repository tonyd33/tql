title: a case with only a default moves into a case scrutinee as a join point of no parameters
grammar: typescript
asserts: simplified, stg, values

`seq` is a case with only a default. Each inner alternative is evaluated,
so each takes the default, which is too large to copy.

--- tql ---
import Data.Function (const);

total xs = case xs of { [] -> 0; h : t -> h + total t };
f b xs = seq (if b then [1] else [2]) (total xs * 2 + total xs * 3 + total xs * 4);
main = const [f True [1], f False [2]];

--- simplified ---
total = \xs -> case xs of
  Nil -> 0
  Cons h t -> op[+] h (total t)
f = \b -> \xs ->
  join j = op[+] (op[+] (op[*] (total xs) 2) (op[*] (total xs) 3)) (op[*] (total xs) 4) in
  case b of
    False -> jump j
    True -> jump j
main = const (Cons (f True (Cons 1 Nil)) (Cons (f False (Cons 2 Nil)) Nil))
