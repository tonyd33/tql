--- tql ---
before :: String -> String -> Bool;
before a b = a < b;

other :: Node -> Node -> Bool;
other m n = m != n;

other_kind :: Node -> Node -> Bool;
other_kind m n = kind m != kind n;

main root = do {
  c <- children root;
  return [before "a" "b", before "b" "a", other root c, other c c, other_kind root c, other_kind c c];
};
