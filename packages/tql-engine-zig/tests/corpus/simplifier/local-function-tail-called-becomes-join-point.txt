title: a local function called only in tail position becomes a join point
grammar: typescript
asserts: simplified, stg, values

`g` is too large to copy into both branches, and each branch ends in a call
to it.

--- tql ---
import Data.Function (const);

f b x = let { g y = y * 2 + y * 3 + y * 4 + y * 5 } in if b then g x else g (x + 1);
main = const [f True 1, f False 1];

f = \b -> \x ->
  join g y = op[+] (op[+] (op[+] (op[*] y 2) (op[*] y 3)) (op[*] y 4)) (op[*] y 5) in
  case b of
    False -> jump g (op[+] x 1)
    True -> jump g x
main = const (Cons (f True 1) (Cons (f False 1) Nil))
