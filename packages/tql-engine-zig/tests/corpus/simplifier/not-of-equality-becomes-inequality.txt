title: negating an equality becomes an inequality
grammar: typescript
asserts: simplified, values

--- tql ---
differ :: Int -> Int -> Bool;
differ x y = not $ x == y;

main = Filter (\_ -> [differ 1 2, differ 2 2]);

--- source ---
class A {}

--- values ---
[true, false]

--- simplified ---
main = \_ -> Cons (%ne_int 1 2) (Cons (%ne_int 2 2) Nil)
