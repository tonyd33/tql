title: negating an equality becomes an inequality
grammar: typescript
asserts: simplified, values

--- tql ---
import Data.Function (const);

differ :: Int -> Int -> Bool;
differ x y = not $ x = y;

main = const [differ 1 2, differ 2 2];

--- source ---
class A {}

--- values ---
[true, false]

--- simplified ---
main =
  let x = Cons (%ne_int 1 2) (Cons (%ne_int 2 2) Nil) in
  \y -> x
