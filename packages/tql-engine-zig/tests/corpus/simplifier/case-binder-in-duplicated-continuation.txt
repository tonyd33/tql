--- tql ---
double :: [Int] -> [Int];
double xs = case xs of {
  [] -> [];
  h : t -> h * 2 : double t;
};
pick :: [Int] -> [Int] -> Int;
pick xs ys = let { zs = case xs of { [] -> ys; _ : _ -> double xs; }; } in case zs of {
  [] -> 0;
  h : _ -> h + length zs + pick [] [];
};
main = pure [pick [] [5, 6], pick [1, 2] [], pick [] []];

double = \xs -> case xs of
  Nil -> Nil
  Cons h t -> Cons (op[*] h 2) (double t)
pick = \xs -> \ys ->
  join j zs h _ = op[+] (op[+] h (length[List] zs)) (pick Nil Nil) in
  case xs of
    Nil -> case ys of
      Nil -> 0
      Cons h _ -> jump j ys h _
    Cons _ _ -> case double xs of zs
      Nil -> 0
      Cons h _ -> jump j zs h _
main =

  \_ -> Cons y Nil
