title: a list pattern is a constructor argument
grammar: typescript
asserts: core, values

--- tql ---
data Maybe a = Just a | Nothing;

only m = case m of {
  Just [x] -> x;
  _ -> 0;
};

main = pure (only (Just [7])) <|> pure (only (Just [7, 8]));

--- source ---
class A {}

--- core ---
only = \m ->
  letrec alternative = 0 in
  case m of
    Just just0 -> case just0 of
      Nil -> alternative
      Cons x cons1 -> case cons1 of
        Nil -> x
        Cons _ _ -> alternative
    Nothing -> alternative
main = alt (pure (only (Just (Cons 7 Nil)))) (pure (only (Just (Cons 7 (Cons 8 Nil)))))
