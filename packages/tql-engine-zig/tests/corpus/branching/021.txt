title: the first alternative that matches wins
grammar: typescript
asserts: core, types, values

`[1]` matches both `Cons x Nil` and `Cons x t`.

--- tql ---
shape xs = case xs of {
  Cons x Nil -> 1;
  Cons x t -> 2;
  Nil -> 0;
};

main = const [shape [1], shape [1, 2], shape []];

--- source ---
class A {}

--- values ---
[1, 2, 0]

--- types ---
shape :: [a] -> Int
main :: Node -> [Int]

--- core ---
shape = \xs -> case xs of
  Nil -> 0
  Cons x t -> case t of
    Nil -> 1
    Cons _ _ -> 2
main = const (Cons (shape (Cons 1 Nil)) (Cons (shape (Cons 1 (Cons 2 Nil))) (Cons (shape Nil) Nil)))
