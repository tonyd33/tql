title: nested patterns match a declared datatype
grammar: typescript
asserts: types, values

A nullary constructor in argument position needs no parentheses.

--- tql ---
type Maybe a = Just a | Nothing;

both :: Maybe Int -> Maybe Int -> Int;
both m n = case Cons m (Cons n Nil) of {
  Cons (Just a) (Cons (Just b) Nil) -> a + b;
  Cons Nothing _ -> 0 - 1;
  _ -> 0;
};

main = const [both (Just 1) (Just 2), both Nothing (Just 2), both (Just 1) Nothing];

--- source ---
class A {}

--- values ---
[3, -1, 0]

--- types ---
both :: Maybe Int -> Maybe Int -> Int
main :: Node -> [Int]
