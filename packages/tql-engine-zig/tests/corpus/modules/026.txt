title: a synonym body reads a function private to its module
grammar: typescript
asserts: simplified, values

--- module Names ---
module Names (pattern Named) for typescript;

name_of n = text n;

pattern Named s <- :class_declaration { #name = (name_of -> s) };

--- tql ---
import Names (pattern Named);

main root = do {
  Named n <- descendants root;
  return n;
};

--- source ---
class A {}
function f() {}

--- values ---
["A"]
--- simplified ---
main = \root -> concat_map (
  \cons0 -> case field[name] cons0 of
    Nil -> Nil
    Cons cons0 cons1 -> case cons1 of
      Nil -> Cons (text cons0) Nil
      Cons _ _ -> Nil
) (descendants_of_kind :class_declaration root)
