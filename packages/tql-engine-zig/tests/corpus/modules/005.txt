data Stack = Nil | Cons Int Stack;
depth s = case s of { Nil -> 0; Cons h t -> 1 + depth t };
size xs = case xs of { [] -> 0; h : t -> 1 + size t };
main root = [size [1, 2, 3], depth (Cons 1 Nil)];
