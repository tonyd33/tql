title: a module written for the query's grammar names its kinds
grammar: typescript
asserts: values

--- module Classes ---
module Classes for typescript, tsx;

class_names = do {
  c <- descendants_of_kind :class_declaration;
  n <- pure c | #name;
  return $ text n;
};

--- tql ---
import Classes;

main = class_names;

--- source ---
class A {}
class B {}

--- values ---
["A", "B"]
