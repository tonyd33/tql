title: imports sharing a qualifier are each reached through it
grammar: typescript
asserts: values

--- module Doubling ---
module Doubling;

f x = x * 2;

--- module Tripling ---
module Tripling;

g x = x * 3;

--- tql ---
import Doubling as Q;
import Tripling as Q;

main = Filter (\_ -> [Q.f 1, Q.g 1]);

--- source ---
let x = 1;

--- values ---
[2, 3]
