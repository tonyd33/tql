title: the kind axes given an anonymous token's kind yield those tokens
grammar: typescript
asserts: values

--- tql ---

main = do {
  root <- identity;
  args@(run children -> open : _) <- descendants_of_kind :arguments;
  return {
    open = text open,
    descendants = map text (run (descendants_of_kind (kind open)) root),
    children = map text (run (children_of_kind (kind open)) args),
  };
};
