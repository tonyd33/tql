title: a kind read off an anonymous token keeps the token through a kind test on an axis
grammar: typescript
asserts: simplified, values

--- tql ---
main root = do {
  open <- descendants root;
  guard $ text open = "(";
  token <- (descendants | of_kind (kind open)) root;
  return $ text token;
};
