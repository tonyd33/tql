title: a kind read off an anonymous token keeps the token through a kind test on an axis
grammar: typescript
asserts: simplified, values

--- tql ---
main root = do {
  open <- descendants root;
  guard $ text open = "(";
  token <- (descendants | of_kind (kind open)) root;
  return $ text token;
};

--- source ---
f(a);

--- values ---
["("]

--- simplified ---
main = \root -> concat_map (
  \open -> concat_map (\_ -> concat_map (\token -> Cons (%text token) Nil) (%descendants_of_kind (%kind open) root)) (
    case op[=] (%text open) "(" of
      False -> Nil
      True -> Cons Unit Nil
  )
) (%descendants root)
