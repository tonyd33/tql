title: a kind read off an anonymous token keeps the token through a kind test on an axis
grammar: typescript
asserts: simplified, values

--- tql ---
main = do {
  open <- descendants;
  guard $ text open == "(";
  token <- descendants | of_kind (kind open);
  return $ text token;
};

--- source ---
f(a);

--- values ---
["("]

--- simplified ---
main = \x -> bind[List] (%descendants x) (
  \y -> bind[List] (
    case %eq_string (%text y) "(" of
      False -> Nil
      True -> Cons () Nil
  ) (\y' -> bind[List] (%descendants_of_kind (%kind y) x) (\y -> Cons (%text y) Nil))
)
