title: a kind read off an anonymous token finds nothing on a named axis
grammar: typescript
asserts: simplified, values

--- tql ---
main root = do {
  open <- descendants root;
  guard $ text open = "(";
  token <- (named_descendants | of_kind (kind open)) root;
  return $ text token;
};

--- source ---

--- simplified ---
main = \root -> bind[List] (%descendants root) (
  \open -> bind[List] (
    case %eq_string (%text open) "(" of
      False -> Nil
      True -> Cons () Nil
  ) (\_ -> bind[List] (bind[List] (%named_descendants root) (of_kind (%kind open))) (\token -> Cons (%text token) Nil))
)
