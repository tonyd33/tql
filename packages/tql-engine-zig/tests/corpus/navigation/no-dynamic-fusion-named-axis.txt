title: a kind read off an anonymous token finds nothing on a named axis
grammar: typescript
asserts: simplified, values

--- tql ---
main = do {
  open <- descendants;
  guard $ text open == "(";
  token <- named_descendants | of_kind (kind open);
  return $ text token;
};

--- source ---
f(a);

--- values ---
[]

--- simplified ---
main = \x -> bind[List] (%descendants x) (
  \y -> bind[List] (
    case %eq_string (%text y) "(" of
      False -> Nil
      True -> Cons () Nil
  ) (
    \y' -> bind[List] (
      bind[List] (%named_descendants x) (
        let k = %kind y in
        \x -> case %is_kind k x of
          False -> Nil
          True -> Cons x Nil
      )
    ) (\y -> Cons (%text y) Nil)
  )
)
