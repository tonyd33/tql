title: an anonymous token is a leaf whose kind is its spelling
grammar: typescript
asserts: types, values

--- tql ---
main = do {
  op <- descendants_of_kind :binary_expression | #operator;
  return {
    kind = kind op,
    children = length (run children op),
    parent = run (parent | arr kind) op,
  };
};
