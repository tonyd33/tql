title: an instance head applies a type to a type that is not a variable
grammar: typescript
asserts: error

--- tql ---
import Data.Function (const);

class Describe a where { describe :: a -> String; };
data Box a = Box a;
instance Describe (Box Int) where { describe b = "box"; };
main = const [];

--- error ---
