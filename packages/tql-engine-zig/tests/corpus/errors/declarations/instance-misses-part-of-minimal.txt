title: an instance names only what it misses of a nested `minimal`
grammar: typescript
asserts: error

`tags` is defined, so only the alternatives are missing. `Bool` defines
nothing, so the whole formula is.

--- tql ---
class Describe a where {
  describe :: a -> String;
  label :: a -> String;
  tags :: a -> [String];
  describe x = label x;
  label x = describe x;
  minimal (describe | label), tags;
};
instance Describe Int where {
  tags n = [];
};
instance Describe Bool where {};
main = pure 1;
