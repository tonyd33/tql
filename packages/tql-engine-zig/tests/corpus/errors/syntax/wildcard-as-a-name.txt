title: `_` is no type, declaration, parameter or label
grammar: typescript
asserts: error

--- tql ---
_ :: Int;
_ = 1;
class C _ where {};
data T _ = T;
type K _ = Int;
r = {_ = 1};
s :: {_: Int} -> Int;
f :: _ -> Int;
g :: {x: Int | _} -> Int;
main = pure 1;

--- error ---

  | ^

error[parse]: `_` is a wildcard and cannot name a definition

  | ^
