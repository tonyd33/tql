title: `guard` in a monad with no `Alternative`
grammar: typescript
asserts: error

`guard` fails by `empty`, which `Done` has no instance for.

--- tql ---
data Done a = Done a;

instance Functor Done where {
  map f d = case d of {
    Done x -> Done (f x);
  };
};

instance Applicative Done where {
  pure = Done;
  ap df dx = case df of {
    Done f -> map f dx;
  };
};

instance Monad Done where {
  bind d f = case d of {
    Done x -> f x;
  };
};

positive = do {
  x <- Done 1;
  guard (x > 0);
  return x;
};

main root = [positive];

--- error ---
error[unsatisfied-constraint]: `Alternative Done` is not satisfied.
  --> 24:3
   |
24 |   guard (x > 0);
   |   ^^^^^
