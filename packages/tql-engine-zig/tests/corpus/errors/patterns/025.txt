title: a module without a for clause cannot use node patterns
grammar: typescript
asserts: error

--- module Classes ---
module Classes;
classes root = do {
  c@:class_declaration {} <- descendants root;
  return c;
};

--- tql ---
import Classes;
main root = classes root;

--- error ---
error[grammar-mismatch]: this module has no `for` clause, so it cannot name the kind `class_declaration`

  |
