title: a literal pattern has the scrutinee's type
grammar: typescript
asserts: error

--- tql ---
import Data.Function (const);

f n = case n + 1 of {
  "one" -> 1;
  _ -> 0;
};

main = const [f 1];

--- error ---
error[type-mismatch]: Expected `Int`, found `String`.
 --> 4:3
  |
4 |   "one" -> 1;
  |   ^^^^^
