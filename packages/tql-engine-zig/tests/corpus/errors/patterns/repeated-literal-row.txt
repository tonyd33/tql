title: a literal row after the same literal is never matched
grammar: typescript
asserts: error

--- tql ---
import Data.Function (const);

f n = case n of {
  1 -> "one";
  1 -> "uno";
  _ -> "other";
};

main = const [f 1];

--- error ---
error[type-mismatch]: `1` is never matched
 --> 5:3
  |
5 |   1 -> "uno";
  |   ^
