title: closures capture their lexical environment
grammar: typescript
asserts: types, values, stg

`bump` captures `step` from the enclosing group, not from its call site.

--- tql ---
import Data.Function (const);
import Data.List (map);

main =
  let {
    step = 5 + 5;
    bump = \x -> x + step;
  }
  in const (map bump [1, 2]);

--- source ---
class A {}

--- values ---
[11, 12]

--- types ---
main :: Node -> [Int]

--- stg ---

      let
        t1 = {step@0} \u {} ->
          let t0 = {step@0} \n {x} -> op[+]# x@1 step@0 in
          t0@1
        t4 = {} \u {} ->
          let
            t3 = {} \u {} ->
              let c2 = Nil in
              Cons 2 c2@0
          in
          Cons 1 t3@0
      in
      map t1@1 t4@2
  in
  const t5@1
