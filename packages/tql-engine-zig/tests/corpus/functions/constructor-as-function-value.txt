title: a constructor with fields is a function value
grammar: typescript
asserts: types, values

--- tql ---
import Data.Function (flip);

data Pair = Pair Int Int;

second :: Pair -> Int;
second p = case p of { Pair _ b -> b; };

main = Filter (\_ -> append
  [second (flip Pair 1 2), length (map (Cons 1) [[2], [3]])]
  (map second (map (Pair 1) [2, 3])));

--- source ---
class A {}

--- values ---
[1, 2, 2, 3]

--- types ---
second :: Pair -> Int
main :: Filter Node Int
