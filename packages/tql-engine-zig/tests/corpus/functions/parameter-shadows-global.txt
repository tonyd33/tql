title: a function parameter shadows a global of the same name
grammar: typescript
asserts: core, types, values

--- tql ---
import Data.Function (const);

n = 100;
shadowed n = n + 1;
main = const [shadowed 1, n];

--- source ---
class A {}

--- values ---
[2, 100]

--- types ---
n :: Int
shadowed :: Int -> Int
main :: Node -> [Int]

--- core ---
n = 100
shadowed = \n -> op[+] n 1
main = const (Cons (shadowed 1) (Cons n Nil))
