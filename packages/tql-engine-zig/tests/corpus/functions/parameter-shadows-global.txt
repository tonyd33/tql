title: a function parameter shadows a global of the same name
grammar: typescript
asserts: core, types, values

--- tql ---
n = 100;
shadowed n = n + 1;
main = Filter (\_ -> [shadowed 1, n]);

--- source ---
class A {}

--- values ---
[2, 100]

--- types ---
n :: Int
shadowed :: Int -> Int
main :: Filter Node Int

--- core ---
n = 100
shadowed = \n -> op[+] n 1
main = Filter (\_ -> Cons (shadowed 1) (Cons n Nil))
