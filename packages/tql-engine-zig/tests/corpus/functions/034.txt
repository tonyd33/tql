title: a constructor with fields is a function value

data Pair = Pair Int Int;

second :: Pair -> Int;
second p = case p of { Pair _ b -> b; };

main = const [second (flip Pair 1 2), length (map (Cons 1) [[2], [3]])];

class A {}
