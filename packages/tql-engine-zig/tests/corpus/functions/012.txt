title: closures capture their lexical environment
grammar: typescript
asserts: types, values, stg

`bump` captures `step` from the enclosing group, not from its call site.

--- tql ---
main =
  let {
    step = 10;
    bump = \x -> x + step;
  }
  in const [bump 1, bump 2];

--- source ---
class A {}

--- values ---
[11, 12]

--- types ---
main :: Node -> [Int]

--- stg ---
main = {} \u {} ->
  let step = {} \u {} -> 10 in
  let bump = {step@0} \n {x} -> op[+]# x@1 step@0 in
  let
    t4 = {bump@1} \u {} ->
      let
        t0 = {bump@0} \u {} -> bump@0 1
        t3 = {bump@0} \u {} ->
          let
            t1 = {bump@0} \u {} -> bump@0 2
            c2 = Nil
          in
          Cons t1@1 c2@2
      in
      Cons t0@1 t3@2
  in
  const t4@2
