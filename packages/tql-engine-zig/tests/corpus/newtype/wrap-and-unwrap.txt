title: a `newtype` wraps and unwraps with no trace in the machine
grammar: typescript
asserts: types, values, simplified, stg

--- tql ---
newtype Name = Name String;

unname :: Name -> String;
unname n = case n of {
  Name s -> s;
};

main = pure $ unname (Name "a");

--- source ---
class A {}

--- values ---
["a"]

--- simplified ---
main =
  let y = "a" in
  \_ -> Cons y Nil
entry = \root -> map[List] JString (main root)

--- stg ---
main = {} \u {} ->
  let y = {} \u {} -> "a" in
  let
    t1 = {y@0} \n {_} ->
      let c0 = Nil in
      Cons y@0 c0@2
  in
  t1@1
entry = {} \u {} ->
  let
    t3 = {} \n {root} ->
      let
        t1 = {} \n {f0} -> JString f0@0
        t2 = {root@0} \u {} -> main root@0
      in
      map[List] t1@1 t2@2
  in
  t3@0

--- types ---
unname :: Name -> String
main :: Kleisli List Node String
entry :: Node -> [Json]
