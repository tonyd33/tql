second = \xs ->
  join alternative = Nil in
  case xs of
    Nil -> jump alternative
    Cons _ cons1 -> case cons1 of
      Nil -> jump alternative
      Cons b _ -> Cons b Nil
main = always (second (Cons 1 (Cons 2 (Cons 3 Nil))))
