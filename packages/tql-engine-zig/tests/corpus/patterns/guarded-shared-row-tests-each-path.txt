title: a guarded row reached from several paths tests its guard on each
grammar: typescript
asserts: core, values, simplified, stg

The second row is reached when the list is empty and when it has two or
more elements. Its body is bound once and its guard is tested on each path.

--- tql ---
import Data.Function (const);
import Data.List (drop, null);

f xs = case xs of {
  [x] -> x;
  ys if not (null (drop 2 ys)) -> 100;
  _ -> 0;
};

main = const [f [5], f [], f [1, 2], f [1, 2, 3]];

f = \xs ->
  join alternative ys = 100 in
  join alternative' = 0 in
  case xs of
    Nil -> case not (null (drop 2 xs)) of
      False -> jump alternative'
      True -> jump alternative xs
    Cons x cons1 -> case cons1 of
      Nil -> x
      Cons _ _ -> case not (null (drop 2 xs)) of
        False -> jump alternative'
        True -> jump alternative xs
main = const (Cons (f (Cons 5 Nil)) (Cons (f Nil) (Cons (f (Cons 1 (Cons 2 Nil))) (Cons (f (Cons 1 (Cons 2 (Cons 3 Nil)))) Nil))))
