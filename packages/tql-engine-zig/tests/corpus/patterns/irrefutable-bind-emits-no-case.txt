title: an irrefutable bind emits no case
grammar: typescript
asserts: core, values

`x`, `x@_` and `_` match every value, so each lowers to a bare `bind`.

--- tql ---
import Data.Function (const);

main = const $ do {
  x <- [1, 2];
  y@_ <- [x];
  _ <- [y];
  return y;
};

--- source ---
class A {}

--- values ---
[1, 2]

--- core ---
main = const (bind[List] (Cons 1 (Cons 2 Nil)) (\x -> bind[List] (Cons x Nil) (\y -> bind[List] (Cons y Nil) (\_ -> return y))))
