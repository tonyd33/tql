title: an as-pattern on a synonym use names the whole value
grammar: typescript
asserts: values

--- tql ---
pattern Callee f <- :call_expression { #function = f };

main root = do {
  c@(Callee f) <- descendants root;
  return [text c, text f];
};

--- source ---
f(1);
g.h(2);

--- values ---
[["f(1)","f"],["g.h(2)","g.h"]]
