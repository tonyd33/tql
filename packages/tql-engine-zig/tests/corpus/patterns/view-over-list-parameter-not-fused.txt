title: an of_kind view bound over a list parameter is not fused
grammar: typescript
asserts: types, simplified, values

--- tql ---
names xs = do {
  (run (of_kind :class_declaration) -> [n]) <- xs;
  return $ text n;
};

main = Filter (\root -> names (run descendants root));

--- source ---
class A {}

--- values ---
["class A {}"]

--- types ---
names :: (Monad a, Alternative a) => a Node -> a String
main :: Filter Node String
--- simplified ---
names = \xs -> bind[List] xs (
  \scrutinee -> case %is_kind :class_declaration scrutinee of
    False -> Nil
    True -> Cons (%text scrutinee) Nil
)
main = \root -> names (%descendants root)
