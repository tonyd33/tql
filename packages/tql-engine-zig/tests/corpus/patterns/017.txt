title: a variable at the top of a conjunction may shadow an earlier bind
grammar: typescript
asserts: core, values

--- tql ---
main = const $ do {
  c <- [1, 2];
  c & 1 <- [c];
  return c;
};

--- source ---
class A {}

--- values ---
[1]

--- core ---
main = const (
  concat_map (
    \c -> concat_map (
      \c -> case op[=] c 1 of
        False -> Nil
        True -> return c
    ) (Cons c Nil)
  ) (Cons 1 (Cons 2 Nil))
)
