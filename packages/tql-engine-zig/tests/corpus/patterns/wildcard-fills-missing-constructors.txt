title: a wildcard alternative covers the constructors before it leaves out
grammar: typescript
asserts: core, types, values

--- tql ---
import Data.Function (const);

first_or_zero xs = case xs of {
  Cons a _ -> a;
  _ -> 0;
};

main = const [first_or_zero [7, 8], first_or_zero []];

--- source ---
class A {}

--- values ---
[7, 0]

--- types ---
first_or_zero :: [Int] -> Int
main :: Node -> [Int]

--- core ---
first_or_zero = \xs -> case xs of
  Nil -> 0
  Cons a _ -> a
main = const (Cons (first_or_zero (Cons 7 (Cons 8 Nil))) (Cons (first_or_zero Nil) Nil))
