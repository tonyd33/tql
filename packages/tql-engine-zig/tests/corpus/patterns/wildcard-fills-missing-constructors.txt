title: a wildcard alternative covers the constructors before it leaves out
grammar: typescript
asserts: core, types, values

--- tql ---
first_or_zero xs = case xs of {
  Cons a _ -> a;
  _ -> 0;
};

main = Filter (\_ -> [first_or_zero [7, 8], first_or_zero []]);

--- source ---
class A {}

--- values ---
[7, 0]

--- types ---
first_or_zero :: [Int] -> Int
main :: Filter Node Int

--- core ---
first_or_zero = \xs -> case xs of
  Nil -> 0
  Cons a _ -> a
main = Filter (\_ -> Cons (first_or_zero (Cons 7 (Cons 8 Nil))) (Cons (first_or_zero Nil) Nil))
