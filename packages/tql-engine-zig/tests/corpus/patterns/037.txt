kids root = do {
  (of_kind :class_declaration -> [n]) <- children root;
  return $ text n;
};

named_kids root = do {
  (of_kind :class_declaration -> [n]) <- named_children root;
  return $ text n;
};

all root = do {
  (of_kind :class_declaration -> [n]) <- descendants root;
  return $ text n;
};

named_all root = do {
  (of_kind :class_declaration -> [n]) <- named_descendants root;
  return $ text n;
};

main = kids <|> named_kids <|> all <|> named_all;
