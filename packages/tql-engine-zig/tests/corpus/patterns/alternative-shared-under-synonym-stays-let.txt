title: an alternative shared under a synonym test stays a let
grammar: typescript
asserts: core, values

The synonym's matcher takes the fallthrough as an argument, so a jump there
would not be a tail call.

--- tql ---
pattern Both a b <- [(text -> a), (text -> b)];

describe xs = case xs of {
  Both "x" "y" -> "xy";
  _ -> "neither";
};

main root = do {
  a <- descendants_of_kind :arguments root;
  return (describe (children_of_kind :identifier a));
};

--- source ---
f(x, y);
f(w, y);

--- values ---
["xy","neither"]
--- core ---
$mBoth = \s -> \k -> \f -> case s of
  Nil -> f
  Cons cons0 cons1 ->
    let a = text cons0 in
    case cons1 of
      Nil -> f
      Cons cons0 cons1 ->
        let b = text cons0 in
        case cons1 of
          Nil -> k a b
          Cons _ _ -> f
describe = \xs ->
  let alternative = "neither" in
  $mBoth xs (
    \both0 -> \both1 -> case eq both0 "x" of
      False -> alternative
      True -> case eq both1 "y" of
        False -> alternative
        True -> "xy"
  ) alternative
main = \root -> concat_map (\a -> return (describe (children_of_kind :identifier a))) (descendants_of_kind :arguments root)
