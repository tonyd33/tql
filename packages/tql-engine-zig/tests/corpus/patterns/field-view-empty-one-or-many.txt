title: a view over a filter matches `[]`, `[d]` or `_ : _` once per node
grammar: typescript
asserts: core, values

A class with two decorators matches `_ : _` once.

--- tql ---
undecorated = do {
  (run #decorator -> []) <- identity;
  return "none";
};

one_decorator = do {
  (run #decorator -> [d]) <- identity;
  return $ text d;
};

decorated = do {
  (run #decorator -> _ : _) <- identity;
  return "some";
};

main = descendants_of_kind :class_declaration | (undecorated <|> one_decorator <|> decorated);

--- source ---
class A {}
@a class B {}
@a @b class C {}

--- values ---
["none", "@a", "some", "some"]

--- core ---
undecorated = bind identity (
  \scrutinee ->
    let view = run (Filter field[decorator]) scrutinee in
    case view of
      Nil -> return "none"
      Cons _ _ -> empty
)
one_decorator = bind identity (
  \scrutinee ->
    let view = run (Filter field[decorator]) scrutinee in
    case view of
      Nil -> empty
      Cons d cons1 -> case cons1 of
        Nil -> return (text d)
        Cons _ _ -> empty
)
decorated = bind identity (
  \scrutinee ->
    let view = run (Filter field[decorator]) scrutinee in
    case view of
      Nil -> empty
      Cons _ _ -> return "some"
)
main = compose_flipped (descendants_of_kind :class_declaration) (alt (alt undecorated one_decorator) decorated)
