title: a view over a filter matches `[]`, `[d]` or `_ : _` once per node
grammar: typescript
asserts: core, values

A class with two decorators matches `_ : _` once.

--- tql ---
undecorated c = do {
  (#decorator -> []) <- [c];
  return "none";
};

one_decorator c = do {
  (#decorator -> [d]) <- [c];
  return $ text d;
};

decorated c = do {
  (#decorator -> _ : _) <- [c];
  return "some";
};

main = descendants_of_kind :class_declaration | (undecorated <|> one_decorator <|> decorated);

--- source ---
class A {}
@a class B {}
@a @b class C {}

--- values ---
["none", "@a", "some", "some"]

--- core ---
undecorated = \c -> bind (Cons c Nil) (
  \scrutinee ->
    let view = field[decorator] scrutinee in
    case view of
      Nil -> return "none"
      Cons _ _ -> empty
)
one_decorator = \c -> bind (Cons c Nil) (
  \scrutinee ->
    let view = field[decorator] scrutinee in
    case view of
      Nil -> empty
      Cons d cons1 -> case cons1 of
        Nil -> return (text d)
        Cons _ _ -> empty
)
decorated = \c -> bind (Cons c Nil) (
  \scrutinee ->
    let view = field[decorator] scrutinee in
    case view of
      Nil -> empty
      Cons _ _ -> return "some"
)
main = kleisli (descendants_of_kind :class_declaration) (union (union undecorated one_decorator) decorated)
