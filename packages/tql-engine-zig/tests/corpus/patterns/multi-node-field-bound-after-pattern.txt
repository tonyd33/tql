title: enumerating a multi-node field is a bind after the pattern
grammar: typescript
asserts: core, values

--- tql ---
main = do {
  (run (of_kind :class_declaration) -> [class_ & (run #name -> [name])]) <- descendants;
  decorator <- pure class_ | #decorator;
  return [text name, text decorator];
};

--- source ---
@a @b class A {}
class B {}
@a class C {}

--- values ---
[["A", "@a"], ["A", "@b"], ["C", "@a"]]

--- core ---
main = bind descendants (
  \scrutinee ->
    let view = run (of_kind :class_declaration) scrutinee in
    case view of
      Nil -> empty
      Cons class_ cons1 ->
        let view = run (Filter field[name]) class_ in
        case view of
          Nil -> empty
          Cons name cons1' -> case cons1' of
            Nil -> case cons1 of
              Nil -> bind (compose_flipped (pure class_) (Filter field[decorator])) (\decorator -> return (Cons (text name) (Cons (text decorator) Nil)))
              Cons _ _ -> empty
            Cons _ _ -> empty
)
