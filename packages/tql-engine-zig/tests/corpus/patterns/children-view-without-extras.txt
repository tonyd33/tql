title: filtering extras lets a positional pattern tolerate comments that break it unfiltered
grammar: typescript
asserts: core, values

The filtered pattern matches both calls. The unfiltered one misses the call
with a comment.

--- tql ---
filtered root = do {
  (of_kind :call_expression -> [
    (#arguments -> [(children | keep (not . is_extra) -> [_, argument, _])])
  ]) <- descendants root;
  return $ text argument;
};

unfiltered root = do {
  (of_kind :call_expression -> [(#arguments -> [(children -> [_, argument, _])])])
    <- descendants root;
  return $ text argument;
};

main = filtered <|> unfiltered;

--- source ---
f(/* c */ a);
g(b);

--- values ---
["a","b","b"]

--- core ---
filtered = \root -> concat_map (
  \scrutinee ->
    let view = of_kind :call_expression scrutinee in
    case view of
      Nil -> Nil
      Cons cons0 cons1 ->
        let view = field[arguments] cons0 in
        case view of
          Nil -> Nil
          Cons cons0 cons1' ->
            let view = kleisli children (keep (compose not is_extra)) cons0 in
            case view of
              Nil -> Nil
              Cons _ cons1'' -> case cons1'' of
                Nil -> Nil
                Cons argument cons1'' -> case cons1'' of
                  Nil -> Nil
                  Cons _ cons1'' -> case cons1'' of
                    Nil -> case cons1' of
                      Nil -> case cons1 of
                        Nil -> return (text argument)
                        Cons _ _ -> Nil
                      Cons _ _ -> Nil
                    Cons _ _ -> Nil
) (descendants root)
unfiltered = \root -> concat_map (
  \scrutinee ->
    let view = of_kind :call_expression scrutinee in
    case view of
      Nil -> Nil
      Cons cons0 cons1 ->
        let view = field[arguments] cons0 in
        case view of
          Nil -> Nil
          Cons cons0 cons1' ->
            let view = children cons0 in
            case view of
              Nil -> Nil
              Cons _ cons1'' -> case cons1'' of
                Nil -> Nil
                Cons argument cons1'' -> case cons1'' of
                  Nil -> Nil
                  Cons _ cons1'' -> case cons1'' of
                    Nil -> case cons1' of
                      Nil -> case cons1 of
                        Nil -> return (text argument)
                        Cons _ _ -> Nil
                      Cons _ _ -> Nil
                    Cons _ _ -> Nil
) (descendants root)
main = alt filtered unfiltered
