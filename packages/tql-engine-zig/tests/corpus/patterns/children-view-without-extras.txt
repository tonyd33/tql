title: filtering extras lets a positional pattern tolerate comments that break it unfiltered
grammar: typescript
asserts: core, values

The filtered pattern matches both calls. The unfiltered one misses the call
with a comment.

--- tql ---
filtered = do {
  (run (of_kind :call_expression) -> [
    (run #arguments -> [(run (children | keep (not . is_extra)) -> [_, argument, _])])
  ]) <- descendants;
  return $ text argument;
};

unfiltered = do {
  (run (of_kind :call_expression) -> [(run #arguments -> [(run children -> [_, argument, _])])])
    <- descendants;
  return $ text argument;
};

main = filtered <|> unfiltered;

--- source ---
f(/* c */ a);
g(b);

--- values ---
["a","b","b"]

--- core ---
filtered = bind descendants (
  \scrutinee ->
    let view = run (of_kind :call_expression) scrutinee in
    case view of
      Nil -> empty
      Cons cons0 cons1 ->
        let view = run (Filter field[arguments]) cons0 in
        case view of
          Nil -> empty
          Cons cons0 cons1' ->
            let view = run (compose_flipped children (keep (compose not is_extra))) cons0 in
            case view of
              Nil -> empty
              Cons _ cons1'' -> case cons1'' of
                Nil -> empty
                Cons argument cons1'' -> case cons1'' of
                  Nil -> empty
                  Cons _ cons1'' -> case cons1'' of
                    Nil -> case cons1' of
                      Nil -> case cons1 of
                        Nil -> return (text argument)
                        Cons _ _ -> empty
                      Cons _ _ -> empty
                    Cons _ _ -> empty
)
unfiltered = bind descendants (
  \scrutinee ->
    let view = run (of_kind :call_expression) scrutinee in
    case view of
      Nil -> empty
      Cons cons0 cons1 ->
        let view = run (Filter field[arguments]) cons0 in
        case view of
          Nil -> empty
          Cons cons0 cons1' ->
            let view = run children cons0 in
            case view of
              Nil -> empty
              Cons _ cons1'' -> case cons1'' of
                Nil -> empty
                Cons argument cons1'' -> case cons1'' of
                  Nil -> empty
                  Cons _ cons1'' -> case cons1'' of
                    Nil -> case cons1' of
                      Nil -> case cons1 of
                        Nil -> return (text argument)
                        Cons _ _ -> empty
                      Cons _ _ -> empty
                    Cons _ _ -> empty
)
main = alt filtered unfiltered
