title: filtering extras lets a positional pattern tolerate comments
grammar: typescript
asserts: core, values

--- tql ---
main root = do {
  (of_kind :call_expression -> [
    (#arguments -> [(children | keep (not . is_extra) -> [_, argument, _])])
  ]) <- descendants root;
  return $ text argument;
};

--- source ---
f(/* c */ a);
g(b);

--- values ---
["a", "b"]

--- core ---
main = \root -> concat_map (
  \scrutinee ->
    let view = of_kind :call_expression scrutinee in
    case view of
      Nil -> Nil
      Cons cons0 cons1 ->
        let view = field[arguments] cons0 in
        case view of
          Nil -> Nil
          Cons cons0 cons1' ->
            let view = kleisli children (keep (compose not is_extra)) cons0 in
            case view of
              Nil -> Nil
              Cons _ cons1'' -> case cons1'' of
                Nil -> Nil
                Cons argument cons1'' -> case cons1'' of
                  Nil -> Nil
                  Cons _ cons1'' -> case cons1'' of
                    Nil -> case cons1' of
                      Nil -> case cons1 of
                        Nil -> return (text argument)
                        Cons _ _ -> Nil
                      Cons _ _ -> Nil
                    Cons _ _ -> Nil
) (descendants root)
