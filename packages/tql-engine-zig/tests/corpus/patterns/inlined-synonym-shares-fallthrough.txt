title: a fallthrough reached from several tests is bound once
grammar: typescript
asserts: simplified, values

--- tql ---
pattern Both a b <- [(text -> a), (text -> b)];

describe xs = case xs of {
  Both "x" "y" -> "xy";
  _ -> "neither";
};

main = do {
  a <- descendants_of_kind :arguments;
  return (describe (run (children_of_kind :identifier) a));
};

--- source ---
f(x, y);
f(x, z);
f(w, y);

--- values ---
["xy","neither","neither"]
--- simplified ---
describe = \xs ->
  join alternative = "neither" in
  case xs of
    Nil -> jump alternative
    Cons cons0 cons1 -> case cons1 of
      Nil -> jump alternative
      Cons cons0' cons1 -> case cons1 of
        Nil -> case %eq_string (%text cons0) "x" of
          False -> jump alternative
          True -> case %eq_string (%text cons0') "y" of
            False -> jump alternative
            True -> "xy"
        Cons _ _ -> jump alternative
main =
  let p = %descendants_of_kind :arguments in
  \x -> bind[List] (p x) (\y -> Cons (describe (%children_of_kind :identifier y)) Nil)
