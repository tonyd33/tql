title: a named node pattern binds the node and its fields
grammar: typescript
asserts: simplified, values

--- tql ---
main root = do {
  c@:class_declaration { #name = n } <- descendants root;
  return [text c, text n];
};

--- source ---
class A {}
class B {}

--- values ---
[["class A {}", "A"], ["class B {}", "B"]]

--- simplified ---
main = \root -> bind[List] (%descendants_of_kind :class_declaration root) (
  \c -> case field[name] c of
    Nil -> Nil
    Cons n cons1 -> case cons1 of
      Nil -> Cons (Cons (%text c) (Cons (%text n) Nil)) Nil
      Cons _ _ -> Nil
)
