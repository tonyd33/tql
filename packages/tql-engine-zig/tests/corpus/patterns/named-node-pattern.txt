title: a named node pattern binds the node and its fields
grammar: typescript
asserts: simplified, values

--- tql ---
main = do {
  c@:class_declaration { #name = n } <- descendants;
  return [text c, text n];
};

--- source ---
class A {}
class B {}

--- values ---
[["class A {}", "A"], ["class B {}", "B"]]

--- simplified ---
main = \x -> bind[List] (%descendants_of_kind :class_declaration x) (
  \y -> case field[name] y of
    Nil -> Nil
    Cons n cons1 -> case cons1 of
      Nil -> Cons (Cons (%text y) (Cons (%text n) Nil)) Nil
      Cons _ _ -> Nil
)
