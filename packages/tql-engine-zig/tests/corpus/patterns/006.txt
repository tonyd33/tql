--- tql ---
main root = do {
  (of_kind :method_definition -> [(#name -> [(text -> name@r"^ng")])]) <- descendants root;
  return name;
};

--- source ---
class A { ngOnInit() {} render() {} ngOnDestroy() {} }

--- values ---
["ngOnInit", "ngOnDestroy"]

--- core ---
main = \root -> concat_map (
  \scrutinee ->
    let view = of_kind :method_definition scrutinee in
    case view of
      Nil -> Nil
      Cons cons0 cons1 ->
        let view = field[name] cons0 in
        case view of
          Nil -> Nil
          Cons cons0 cons1' ->
            let name = text cons0 in
            case op[~] name r"^ng" of
              False -> Nil
              True -> case cons1' of
                Nil -> case cons1 of
                  Nil -> return name
                  Cons _ _ -> Nil
                Cons _ _ -> Nil
) (descendants root)
