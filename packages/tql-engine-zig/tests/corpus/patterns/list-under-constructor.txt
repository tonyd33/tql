title: a list pattern is a constructor argument
grammar: typescript
asserts: core, values

--- tql ---
data Maybe a = Just a | Nothing;

only m = case m of {
  Just [x] -> x;
  _ -> 0;
};

main = always (only (Just [7])) <|> always (only (Just [7, 8]));

--- source ---
class A {}

--- core ---
only = \m ->
  join alternative = 0 in
  case m of
    Just just0 -> case just0 of
      Nil -> jump alternative
      Cons x cons1 -> case cons1 of
        Nil -> x
        Cons _ _ -> jump alternative
    Nothing -> jump alternative
main = union (always (only (Just (Cons 7 Nil)))) (always (only (Just (Cons 7 (Cons 8 Nil)))))
