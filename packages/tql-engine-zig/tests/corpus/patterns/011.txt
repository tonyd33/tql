title: a view may use a variable bound to its left in the same pattern
grammar: typescript
asserts: core, values

--- tql ---
main = const $ do {
  [n, (\x -> x + n -> 3)] <- [[1, 2], [2, 2], [0, 3]];
  return n;
};

--- source ---
class A {}

--- values ---
[1, 0]

--- core ---
main = const (
  concat_map (
    \scrutinee -> case scrutinee of
      Nil -> Nil
      Cons n cons1 -> case cons1 of
        Nil -> Nil
        Cons cons0 cons1 ->
          let view = (\x -> op[+] x n) cons0 in
          case eq view 3 of
            False -> Nil
            True -> case cons1 of
              Nil -> return n
              Cons _ _ -> Nil
  ) (Cons (Cons 1 (Cons 2 Nil)) (Cons (Cons 2 (Cons 2 Nil)) (Cons (Cons 0 (Cons 3 Nil)) Nil)))
)
