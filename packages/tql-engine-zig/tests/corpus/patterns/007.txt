title: a field view reaches an anonymous operator token
grammar: typescript
asserts: core, values

--- tql ---
main root = do {
  (of_kind :assignment_expression -> [
    assignment
      & (#left -> [target])
      & (#right -> [(of_kind :binary_expression -> [
          (#left -> [operand]) & (#operator -> [(text -> "||")])
        ])])
  ]) <- descendants root;
  guard $ text target = text operand;
  return $ text assignment;
};

--- source ---
x = x || y;
x = z || y;
x = x && y;

--- values ---
["x = x || y"]

--- core ---
main = \root -> concat_map (
  \scrutinee ->
    let view = of_kind :assignment_expression scrutinee in
    case view of
      Nil -> Nil
      Cons assignment cons1 ->
        let view = field[left] assignment in
        case view of
          Nil -> Nil
          Cons target cons1' -> case cons1' of
            Nil ->
              let view = field[right] assignment in
              case view of
                Nil -> Nil
                Cons cons0 cons1' ->
                  let view = of_kind :binary_expression cons0 in
                  case view of
                    Nil -> Nil
                    Cons cons0 cons1'' ->
                      let view = field[left] cons0 in
                      case view of
                        Nil -> Nil
                        Cons operand cons1''' -> case cons1''' of
                          Nil ->
                            let view = field[operator] cons0 in
                            case view of
                              Nil -> Nil
                              Cons cons0 cons1''' ->
                                let view = text cons0 in
                                case op[=] view "||" of
                                  False -> Nil
                                  True -> case cons1''' of
                                    Nil -> case cons1'' of
                                      Nil -> case cons1' of
                                        Nil -> case cons1 of
                                          Nil -> concat_map (\_ -> return (text assignment)) (guard (op[=] (text target) (text operand)))
                                          Cons _ _ -> Nil
                                        Cons _ _ -> Nil
                                      Cons _ _ -> Nil
                                    Cons _ _ -> Nil
                          Cons _ _ -> Nil
            Cons _ _ -> Nil
) (descendants root)
