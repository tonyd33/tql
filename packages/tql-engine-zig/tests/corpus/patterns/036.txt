title: a do bind skips the values a literal row of its pattern rejects
grammar: typescript
asserts: core, values

--- tql ---
main root = do {
  m@(#name -> [(text -> "render")]) <- descendants_of_kind :method_definition root;
  return $ text m;
};

--- source ---
class A { render() {} update() {} }

--- values ---
["render() {}"]

--- core ---
main = \root -> concat_map (
  \m ->
    let view = field[name] m in
    case view of
      Nil -> Nil
      Cons cons0 cons1 ->
        let view = text cons0 in
        case op[=] view "render" of
          False -> Nil
          True -> case cons1 of
            Nil -> return (text m)
            Cons _ _ -> Nil
) (descendants_of_kind :method_definition root)
