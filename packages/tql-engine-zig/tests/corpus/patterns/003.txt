title: a view over a filter matches its whole output once
grammar: typescript
asserts: core, values

--- tql ---
main root = do {
  (of_kind :class_declaration -> [(#decorator -> _ : _) & (#name -> [(text -> name)])])
    <- descendants root;
  return name;
};

--- source ---
@a @b class A {}
class B {}
@a class C {}

--- values ---
["A", "C"]

--- core ---
main = \root -> concat_map (
  \scrutinee ->
    let view = of_kind :class_declaration scrutinee in
    case view of
      Nil -> Nil
      Cons cons0 cons1 ->
        let view = field[decorator] cons0 in
        case view of
          Nil -> Nil
          Cons _ _ ->
            let view = field[name] cons0 in
            case view of
              Nil -> Nil
              Cons cons0 cons1' ->
                let name = text cons0 in
                case cons1' of
                  Nil -> case cons1 of
                    Nil -> return name
                    Cons _ _ -> Nil
                  Cons _ _ -> Nil
) (descendants root)
