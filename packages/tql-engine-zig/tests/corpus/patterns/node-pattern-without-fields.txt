title: a kind with no fields matches every node of that kind
grammar: typescript
asserts: simplified, values

--- tql ---
main = do {
  :class_declaration {} <- descendants;
  return "class";
};

--- source ---
class A {}
function f() { class B {} }

--- values ---
["class", "class"]

--- simplified ---
main = \x -> bind[List] (%descendants_of_kind :class_declaration x) (\y -> Cons "class" Nil)
