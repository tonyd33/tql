title: a kind with no fields matches every node of that kind
grammar: typescript
asserts: simplified, values

--- tql ---
main root = do {
  :class_declaration {} <- descendants root;
  return "class";
};

--- source ---
class A {}
function f() { class B {} }

--- values ---
["class", "class"]

--- simplified ---
main = \root -> bind[List] (%descendants_of_kind :class_declaration root) (\_ -> Cons "class" Nil)
