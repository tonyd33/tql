title: a top-level as-pattern may shadow an earlier bind
grammar: typescript
asserts: core, values

--- tql ---
main = Filter (\_ -> do {
  x <- [[1]];
  x@[y] <- [[2], [3, 4]];
  return [x, [y]];
});

--- source ---
class A {}

--- values ---
[[[2], [2]]]

--- core ---
main = Filter (
  \_ -> bind (Cons (Cons 1 Nil) Nil) (
    \x -> bind (Cons (Cons 2 Nil) (Cons (Cons 3 (Cons 4 Nil)) Nil)) (
      \x -> case x of
        Nil -> empty
        Cons y cons1 -> case cons1 of
          Nil -> return (Cons x (Cons (Cons y Nil) Nil))
          Cons _ _ -> empty
    )
  )
)
