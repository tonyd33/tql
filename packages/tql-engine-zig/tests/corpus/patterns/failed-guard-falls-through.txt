title: a guard that fails falls through to the next row
grammar: typescript
asserts: core, values, simplified, stg

`[1, 2]` matches the first row's pattern, its guard is false, and the second
row matches. The last row is reached from several paths and is bound once.

--- tql ---
import Data.Function (const);

f xs = case xs of {
  [a, b] if a = b -> "pair";
  [_, _] -> "two";
  _ -> "other";
};

main = const [f [1, 1], f [1, 2], f [1]];

--- source ---
class A {}

--- values ---
["pair", "two", "other"]

--- core ---
f = \xs ->
  join alternative = "other" in
  case xs of
    Nil -> jump alternative
    Cons a cons1 -> case cons1 of
      Nil -> jump alternative
      Cons b cons1 -> case cons1 of
        Nil -> case eq a b of
          False -> "two"
          True -> "pair"
        Cons _ _ -> jump alternative
main = const (Cons (f (Cons 1 (Cons 1 Nil))) (Cons (f (Cons 1 (Cons 2 Nil))) (Cons (f (Cons 1 Nil)) Nil)))
