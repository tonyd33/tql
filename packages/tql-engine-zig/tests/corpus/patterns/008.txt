title: a constructor pattern skips values built by another constructor
grammar: typescript
asserts: core, values

--- tql ---
main = const $ do {
  Cons h _ <- [[1, 2], [], [3]];
  return h;
};

--- source ---
class A {}

--- values ---
[1, 3]

--- core ---
main = const (
  concat_map (
    \scrutinee -> case scrutinee of
      Nil -> Nil
      Cons h _ -> return h
  ) (Cons (Cons 1 (Cons 2 Nil)) (Cons Nil (Cons (Cons 3 Nil) Nil)))
)
