--- core ---
hook = \m ->
  let alternative = Nil in
  let view = field[name] m in
  case view of
    Nil -> alternative
    Cons cons0 cons1 ->
      let name = text cons0 in
      case cons1 of
        Nil -> case op[~] name r"^ng" of
          False -> alternative
          True -> Cons name Nil
        Cons _ _ -> alternative
main = kleisli (descendants_of_kind :method_definition) hook
