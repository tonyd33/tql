--- core ---
pair = \xs ->
  join alternative = Nil in
  case xs of
    Nil -> jump alternative
    Cons a cons1 -> case cons1 of
      Nil -> jump alternative
      Cons b cons1 -> case cons1 of
        Nil -> Cons a (Cons b Nil)
        Cons _ _ -> jump alternative
main = alt (pure (pair (Cons 1 (Cons 2 Nil)))) (pure (pair (Cons 1 (Cons 2 (Cons 3 Nil)))))
