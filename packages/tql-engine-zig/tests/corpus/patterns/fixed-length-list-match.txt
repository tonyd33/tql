title: a list pattern matches a fixed length
grammar: typescript
asserts: core, types, values

--- tql ---
pair xs = case xs of {
  [a, b] -> [a, b];
  _ -> [];
};

main = pure (pair [1, 2]) <|> pure (pair [1, 2, 3]);

--- source ---
class A {}

--- values ---
[[1,2],[]]
--- types ---
pair :: [a] -> [a]
main :: Node -> [[Int]]

--- core ---
pair = \xs ->
  let alternative = Nil in
  case xs of
    Nil -> alternative
    Cons a cons1 -> case cons1 of
      Nil -> alternative
      Cons b cons1 -> case cons1 of
        Nil -> Cons a (Cons b Nil)
        Cons _ _ -> alternative
main = alt (pure (pair (Cons 1 (Cons 2 Nil)))) (pure (pair (Cons 1 (Cons 2 (Cons 3 Nil)))))
