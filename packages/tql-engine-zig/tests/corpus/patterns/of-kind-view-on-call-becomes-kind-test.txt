title: a case of of_kind on a call's result becomes a kind test on the call bound once
grammar: typescript
asserts: simplified, values, stg

Inlining `of_kind` binds its node operand, so `top c` is evaluated once for
both the kind test and `n`.

--- tql ---
top n = case parent n of { [p] -> top p; _ -> n };

main root = do {
  c <- children root;
  [
    case of_kind :class_declaration (top c) of { [n] -> text n; _ -> "other" },
    case of_kind :class_declaration c of { [n] -> text n; _ -> "other" },
  ];
};

--- source ---
class A {}

--- values ---
["other", "class A {}"]

--- simplified ---
top = \n -> case %parent n of
  Nil -> n
  Cons p cons1 -> case cons1 of
    Nil -> top p
    Cons _ _ -> n
main = \root -> concat_map (
  \c -> Cons (
    let x = top c in
    case %is_kind :class_declaration x of
      False -> "other"
      True -> %text x
  ) (
    Cons (
      case %is_kind :class_declaration c of
        False -> "other"
        True -> %text c
    ) Nil
  )
) (%children root)

--- stg ---
