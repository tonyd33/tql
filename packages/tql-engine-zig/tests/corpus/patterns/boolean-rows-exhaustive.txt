grammar: typescript
asserts: core, values

--- tql ---
f b = case b of { True -> "yes"; False -> "no"; };

main = Kleisli $ \_ -> [f True, f False];

--- source ---
class A {}

--- values ---
["yes", "no"]

--- core ---
f = \b -> case b of
  False -> "no"
  True -> "yes"
main = Kleisli (\_ -> Cons (f True) (Cons (f False) Nil))
