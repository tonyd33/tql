--- tql ---
import Data.Function (const);

main = const $ do {
  [a, b] <- [[1, 2], [3], [4, 5], [6, 7, 8]];
  h : _ <- [[a + b], []];
  return h;
};

--- core ---
main = const (
  bind (Cons (Cons 1 (Cons 2 Nil)) (Cons (Cons 3 Nil) (Cons (Cons 4 (Cons 5 Nil)) (Cons (Cons 6 (Cons 7 (Cons 8 Nil))) Nil)))) (
    \scrutinee -> case scrutinee of
      Nil -> empty
      Cons a cons1 -> case cons1 of
        Nil -> empty
        Cons b cons1 -> case cons1 of
          Nil -> bind (Cons (Cons (op[+] a b) Nil) (Cons Nil Nil)) (
            \scrutinee -> case scrutinee of
              Nil -> empty
              Cons h _ -> return h
          )
          Cons _ _ -> empty
  )
)
