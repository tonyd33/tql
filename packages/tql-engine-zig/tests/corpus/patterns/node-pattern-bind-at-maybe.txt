title: a node pattern bound at `Maybe` yields `Nothing` where it does not match
grammar: typescript
asserts: types, core, simplified, values

The pattern tests the node the same way in any monad: `of_kind` and the
field read each give a list, matched as one element. Each failed test yields
`empty`, which is `Nothing` at `Maybe`. The bind is not over an axis, so
nothing fuses.

--- tql ---
import Data.Maybe (Maybe(..));

name_of m = do {
  :class_declaration { #name = n } <- m;
  return $ text n;
};

main root = append (map (\c -> name_of (Just c)) (children root)) [name_of (Just root), name_of Nothing];

name_of = \m -> bind m (
  \scrutinee ->
    let view = of_kind :class_declaration scrutinee in
    case view of

          Nil -> empty

--- simplified ---
name_of = \m -> case m of
  Nothing -> Nothing
  Just x -> case %is_kind :class_declaration x of
    False -> Nothing
    True -> case field[name] x of
      Nil -> Nothing
      Cons n cons1 -> case cons1 of
        Nil -> Just (%text n)
        Cons _ _ -> Nothing
main = \root -> append[List] (map[List] (\c -> name_of (Just c)) (%children root)) (Cons (name_of (Just root)) (Cons Nothing Nil))
