title: a synonym nested three deep inlines flat
grammar: typescript
asserts: core, simplified, values

--- tql ---
pattern Callee f <- :call_expression { #function = f };
pattern Method o p <- Callee (:member_expression { #object = o, #property = p });
pattern ConsoleCall p <- Method (:identifier {} & (text -> "console")) p;

main root = do {
  ConsoleCall p <- descendants root;
  return (text p);
};

--- source ---
console.error("a");
console.log("b");
foo.warn("c");

--- values ---
["error","log"]
--- simplified ---
main = \root -> bind[List] (%descendants_of_kind :call_expression root) (
  \scrutinee -> case field[function] scrutinee of
    Nil -> Nil
    Cons f cons1 -> case cons1 of
      Nil -> case %is_kind :member_expression f of
        False -> Nil
        True -> case field[object] f of
          Nil -> Nil
          Cons o cons1 -> case cons1 of
            Nil -> case field[property] f of
              Nil -> Nil
              Cons p cons1 -> case cons1 of
                Nil -> case %is_kind :identifier o of
                  False -> Nil
                  True -> case %eq_string (%text o) "console" of
                    False -> Nil
                    True -> Cons (%text p) Nil
                Cons _ _ -> Nil
            Cons _ _ -> Nil
      Cons _ _ -> Nil
)

--- core ---
$mCallee = \s -> \k -> \f ->
  let view = of_kind :call_expression s in
  case view of
    Nil -> f
    Cons cons0 cons1 ->
      let view = field[function] cons0 in
      case view of
        Nil -> f
        Cons f' cons1' -> case cons1' of
          Nil -> case cons1 of
            Nil -> k f'
            Cons _ _ -> f
          Cons _ _ -> f
$mMethod = \s -> \k -> \f -> $mCallee s (
  \callee0 ->
    let view = of_kind :member_expression callee0 in
    case view of
      Nil -> f
      Cons cons0 cons1 ->
        let view = field[object] cons0 in
        case view of
          Nil -> f
          Cons o cons1' -> case cons1' of
            Nil ->
              let view = field[property] cons0 in
              case view of
                Nil -> f
                Cons p cons1' -> case cons1' of
                  Nil -> case cons1 of
                    Nil -> k o p
                    Cons _ _ -> f
                  Cons _ _ -> f
            Cons _ _ -> f
) f
$mConsoleCall = \s -> \k -> \f -> $mMethod s (
  \method0 -> \p ->
    let view = of_kind :identifier method0 in
    case view of
      Nil -> f
      Cons cons0 cons1 ->
        let view = text cons0 in
        case eq view "console" of
          False -> f
          True -> case cons1 of
            Nil -> k p
            Cons _ _ -> f
) f
main = \root -> bind (descendants root) (\scrutinee -> $mConsoleCall scrutinee (\p -> return (text p)) empty)
