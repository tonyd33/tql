title: a bind fused onto an axis still reads the variable naming the whole node
grammar: typescript
asserts: simplified, values

--- tql ---
main root = do {
  s & (of_kind :class_declaration -> [n]) <- descendants root;
  return [text s, text n];
};

class A {}

--- values ---
[["class A {}", "class A {}"]]

--- simplified ---
