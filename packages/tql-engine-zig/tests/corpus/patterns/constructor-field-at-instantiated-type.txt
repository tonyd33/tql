title: a case binds a constructor's field at its instantiated type
grammar: typescript
asserts: types, values

--- tql ---
data Maybe a = Just a | Nothing;
or_default m d = case m of { Just x -> x; Nothing -> d };
main = always $ or_default (Just 42) 0;

--- source ---
class A {}

--- values ---
[42]

--- types ---
or_default :: Maybe a -> a -> a
main :: Node -> [Int]
