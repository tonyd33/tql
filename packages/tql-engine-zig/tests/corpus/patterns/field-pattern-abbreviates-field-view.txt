});

  (run #function -> [f]) -> [f];

--- simplified ---
sugar = \n -> case field[function] n of
  Nil -> Nil
  Cons f cons1 -> case cons1 of
    Nil -> Cons f Nil
    Cons _ _ -> Nil
written = \n -> case field[function] n of
  Nil -> Nil
  Cons f cons1 -> case cons1 of
    Nil -> Cons f Nil
    Cons _ _ -> Nil
main = \x -> bind[List] (%descendants x) (\x -> append[List] (sugar x) (written x))
