title: a view pattern tests a node's shape through its fields and text
grammar: typescript
asserts: core, simplified, values

--- tql ---
import Data.Foldable (null);

main = do {
  (run (of_kind :call_expression) -> [
    call & (run #function -> [(run (of_kind :member_expression) -> [
      (run #object -> [(text -> "console")]) & (run #property -> [(text -> "error")])
    ])])
  ]) <- descendants;
  guard $ null $ run (ancestors | of_kind :catch_clause) call;
  return $ text call;
};

--- source ---
console.error('a');
try { console.error('t'); } catch (e) { console.error(e); }
console.warn('b');

--- values ---
["console.error('a')", "console.error('t')"]

--- core ---
main = bind descendants (
  \scrutinee ->
    let view = run (of_kind :call_expression) scrutinee in
    case view of
      Nil -> empty
      Cons call cons1 ->
        let view = run (Filter field[function]) call in
        case view of
          Nil -> empty
          Cons cons0 cons1' ->
            let view = run (of_kind :member_expression) cons0 in
            case view of
              Nil -> empty
              Cons cons0 cons1'' ->
                let view = run (Filter field[object]) cons0 in
                case view of
                  Nil -> empty
                  Cons cons0' cons1''' ->
                    let view = text cons0' in
                    case eq view "console" of
                      False -> empty
                      True -> case cons1''' of
                        Nil ->
                          let view = run (Filter field[property]) cons0 in
                          case view of
                            Nil -> empty
                            Cons cons0 cons1''' ->
                              let view = text cons0 in
                              case eq view "error" of
                                False -> empty
                                True -> case cons1''' of
                                  Nil -> case cons1'' of
                                    Nil -> case cons1' of
                                      Nil -> case cons1 of
                                        Nil -> bind (guard (null (run (compose_flipped ancestors (of_kind :catch_clause)) call))) (\_ -> return (text call))
                                        Cons _ _ -> empty
                                      Cons _ _ -> empty
                                    Cons _ _ -> empty
                                  Cons _ _ -> empty
                        Cons _ _ -> empty
)

--- simplified ---
main = \x -> bind[List] (%descendants_of_kind :call_expression x) (
  \y -> case field[function] y of
    Nil -> Nil
    Cons cons0 cons1 -> case %is_kind :member_expression cons0 of
      False -> Nil
      True -> case field[object] cons0 of
        Nil -> Nil
        Cons cons0' cons1' -> case %eq_string (%text cons0') "console" of
          False -> Nil
          True -> case cons1' of
            Nil -> case field[property] cons0 of
              Nil -> Nil
              Cons cons0 cons1' -> case %eq_string (%text cons0) "error" of
                False -> Nil
                True -> case cons1' of
                  Nil -> case cons1 of
                    Nil -> bind[List] (
                      case (
                        bind[List] (%ancestors y) (
                          \x -> case %is_kind :catch_clause x of
                            False -> Nil
                            True -> Cons x Nil
                        )
                      ) of
                        Nil -> Cons () Nil
                        Cons h t -> Nil
                    ) (\y' -> Cons (%text y) Nil)
                    Cons _ _ -> Nil
                  Cons _ _ -> Nil
            Cons _ _ -> Nil
)
