title: a view pattern tests a node's shape through its fields and text
grammar: typescript
asserts: core, simplified, values

--- tql ---
import Data.Foldable (null);

main root = do {
  (of_kind :call_expression -> [
    call & (#function -> [(of_kind :member_expression -> [
      (#object -> [(text -> "console")]) & (#property -> [(text -> "error")])
    ])])
  ]) <- descendants root;
  guard $ null $ (ancestors | of_kind :catch_clause) call;
  return $ text call;
};

--- source ---
console.error('a');
try { console.error('t'); } catch (e) { console.error(e); }
console.warn('b');

--- values ---
["console.error('a')", "console.error('t')"]

--- core ---
main = \root -> bind (descendants root) (
  \scrutinee ->
    let view = of_kind :call_expression scrutinee in
    case view of
      Nil -> empty
      Cons call cons1 ->
        let view = field[function] call in
        case view of
          Nil -> empty
          Cons cons0 cons1' ->
            let view = of_kind :member_expression cons0 in
            case view of
              Nil -> empty
              Cons cons0 cons1'' ->
                let view = field[object] cons0 in
                case view of
                  Nil -> empty
                  Cons cons0' cons1''' ->
                    let view = text cons0' in
                    case eq view "console" of
                      False -> empty
                      True -> case cons1''' of
                        Nil ->
                          let view = field[property] cons0 in
                          case view of
                            Nil -> empty
                            Cons cons0 cons1''' ->
                              let view = text cons0 in
                              case eq view "error" of
                                False -> empty
                                True -> case cons1''' of
                                  Nil -> case cons1'' of
                                    Nil -> case cons1' of
                                      Nil -> case cons1 of
                                        Nil -> bind (guard (null (kleisli ancestors (of_kind :catch_clause) call))) (\_ -> return (text call))
                                        Cons _ _ -> empty
                                      Cons _ _ -> empty
                                    Cons _ _ -> empty
                                  Cons _ _ -> empty
                        Cons _ _ -> empty
)

--- simplified ---
main = \root -> bind[List] (%descendants_of_kind :call_expression root) (
  \scrutinee -> case field[function] scrutinee of
    Nil -> Nil
    Cons cons0 cons1 -> case %is_kind :member_expression cons0 of
      False -> Nil
      True -> case field[object] cons0 of
        Nil -> Nil
        Cons cons0' cons1' -> case %eq_string (%text cons0') "console" of
          False -> Nil
          True -> case cons1' of
            Nil -> case field[property] cons0 of
              Nil -> Nil
              Cons cons0 cons1' -> case %eq_string (%text cons0) "error" of
                False -> Nil
                True -> case cons1' of
                  Nil -> case cons1 of
                    Nil -> bind[List] (
                      case bind[List] (%ancestors scrutinee) (of_kind :catch_clause) of
                        Nil -> Cons Unit Nil
                        Cons h t -> Nil
                    ) (\_ -> Cons (%text scrutinee) Nil)
                    Cons _ _ -> Nil
                  Cons _ _ -> Nil
            Cons _ _ -> Nil
)
