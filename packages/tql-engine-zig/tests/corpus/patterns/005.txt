title: a view whose pattern binds nothing still matches once
grammar: typescript
asserts: core, values

--- tql ---
main root = do {
  (of_kind :method_definition -> [
    (ancestors | of_kind :class_declaration -> _ : _) & (#name -> [(text -> name)])
  ]) <- descendants root;
  return name;
};

--- source ---
class A { m() { class B { n() {} } } }

--- values ---
["m", "n"]

--- core ---
main = \root -> concat_map (
  \scrutinee ->
    let view = of_kind :method_definition scrutinee in
    case view of
      Nil -> Nil
      Cons cons0 cons1 ->
        let view = kleisli ancestors (of_kind :class_declaration) cons0 in
        case view of
          Nil -> Nil
          Cons _ _ ->
            let view = field[name] cons0 in
            case view of
              Nil -> Nil
              Cons cons0 cons1' ->
                let name = text cons0 in
                case cons1' of
                  Nil -> case cons1 of
                    Nil -> return name
                    Cons _ _ -> Nil
                  Cons _ _ -> Nil
) (descendants root)
