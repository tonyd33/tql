title: a record holding a collected list
grammar: typescript
asserts: core, types, values

Collection inside a field keeps the record single-valued: the list is one
value, so no product fan-out occurs. A class with no methods still yields a
record, with an empty list.

--- tql ---
main = do {
  c <- children_of_kind :class_declaration;
  n <- pure c | #name;
  methods <- pure c | collect (#body | children | of_kind :method_definition | #name | arr text);
  return { name = text n, methods = methods };
};

--- source ---
class A { m() {} n() {} }
class B {}

--- values ---
[
  { "methods": ["m", "n"], "name": "A" },
  { "methods": [], "name": "B" }
]

--- types ---
main :: Filter Node {methods: [String], name: String}

--- core ---
main = bind (children_of_kind :class_declaration) (\c -> bind (compose_flipped (pure c) (Filter field[name])) (\n -> bind (compose_flipped (pure c) (collect (compose_flipped (compose_flipped (compose_flipped (compose_flipped (Filter field[body]) children) (of_kind :method_definition)) (Filter field[name])) (arr text)))) (\methods -> return (record[methods,name] methods (text n)))))
