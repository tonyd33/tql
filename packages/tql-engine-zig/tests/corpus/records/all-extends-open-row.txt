outer :: All Eq r => {name: String | r} -> {name: String | r} -> Bool;

main = pure [outer {name = "a", m = 1, z = 2} {name = "a", m = 1, z = 2}, outer {name = "a", m = 1, z = 2} {name = "b", m = 1, z = 2}];
