title: records nest and hold heterogeneous fields
grammar: typescript
asserts: core, types, values

--- tql ---
main root = do {
  c <- children_of_kind :class_declaration root;
  n <- c#name;
  return { name = text n, meta = { k = kind c, n = 1 } };
};

--- source ---
class A {}

--- values ---
[{ "meta": { "k": "class_declaration", "n": 1 }, "name": "A" }]

--- types ---
main :: Node -> [{meta: {k: Kind, n: Int}, name: String}]

--- core ---
main = \root -> bind[List] (children_of_kind :class_declaration root) (\c -> bind[List] (field[name] c) (\n -> return (record[meta,name] (record[k,n] (kind c) 1) (text n))))
