title: records nest and hold heterogeneous fields
grammar: typescript
asserts: core, types, values

--- tql ---
main = do {
  c <- children_of_kind :class_declaration;
  n <- pure c | #name;
  return { name = text n, meta = { k = kind c, n = 1 } };
};

--- source ---
class A {}

--- values ---
[{ "meta": { "k": "class_declaration", "n": 1 }, "name": "A" }]

--- types ---
main :: Filter Node {meta: {k: Kind, n: Int}, name: String}

--- core ---
main = bind (children_of_kind :class_declaration) (\c -> bind (compose_flipped (pure c) (Filter field[name])) (\n -> return (record[meta,name] (record[k,n] (kind c) 1) (text n))))
