title: `keep`, `none`, `union` and `guard` work at a query's own `Alternative`
grammar: typescript
asserts: types, values

`Option`'s `alt` takes the first success, so `union` falls back.

--- tql ---
import Data.List (concat_map);
import Data.Filter (union);

data Option a = None | Some a;

instance Functor Option where {
  map f o = case o of {
    None -> None;
    Some x -> Some (f x);
  };
};

instance Applicative Option where {
  pure = Some;
  ap af ox = case af of {
    None -> None;
    Some f -> map f ox;
  };
};

instance Alternative Option where {
  empty = None;
  alt a b = case a of {
    None -> b;
    Some x -> Some x;
  };
};

to_list o = case o of {
  None -> [];
  Some x -> [x];
};

checks :: [Option Int];
checks = [
  keep (> 1) 2,
  keep (> 1) 0,
  none 0,
  union (keep (> 5)) (always 7) 3,
  map (\_ -> 9) (guard True),
];

main root = concat_map to_list checks;
