title: a let-bound function using a method is used at two types
grammar: typescript
asserts: values

--- tql ---
import Data.List (append);

class Describe a where {
  describe :: a -> String;
};

instance Describe Int where {
  describe n = "int";
};

instance Describe String where {
  describe s = "string";
};

main root = let { show x = [describe x, describe x] } in append (show 1) (show "a");

class A {}
