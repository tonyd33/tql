import Data.List (map);

class Scale a where {
  scale :: a -> Int -> Int;
};

data Unit = Single | Double;

instance Scale Unit where {
  scale u = case u of { Single -> \n -> n; Double -> \n -> n * 2 };
};

main root = map (scale Double) [1, 2, 3];
