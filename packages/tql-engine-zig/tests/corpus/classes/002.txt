title: a signature names a user class in its context
grammar: typescript
asserts: types

--- tql ---
class Describe a where {
  describe :: a -> String;
};

instance Describe Int where {
  describe n = "int";
};

pretty :: Describe a => a -> [String];
pretty x = [describe x];

main root = pretty 1;

class A {}
