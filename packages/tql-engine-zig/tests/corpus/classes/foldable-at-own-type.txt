--- tql ---
import Data.Foldable (fold_map, fold, to_list, null, any, all, concat);

data Pair a = Pair a a;

instance Foldable Pair where {
  foldr f z p = case p of {
    Pair x y -> f x $ f y z;
  };
};

flag b = if b then 1 else 0;

main = Filter (\_ -> concat [
  to_list $ Pair 1 2,
  fold_map (\x -> [x]) $ Pair 3 4,
  fold $ Pair [5] [6],
  concat $ Pair [7] [8],
  map flag [null $ Pair 0 0, any (= 7) $ Pair 6 7, all (= 7) $ Pair 6 7],
]);
