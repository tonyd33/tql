title: a class, its instances and a constrained function
grammar: typescript
asserts: types, values

`twice` raises `Describe a` twice and takes it once.

--- tql ---
class Eq a => Describe a where {
  describe :: a -> String;
};

data Shape = Circle Int | Square Int deriving (Eq);

instance Describe Shape where {
  describe s = case s of { Circle _ -> "circle"; Square _ -> "square" };
};

instance Describe a => Describe [a] where {
  describe xs = case xs of { [] -> "empty"; x : _ -> describe x };
};

twice x = [describe x, describe x];

main root = twice [Circle 1];

class A {}

--- values ---
["circle", "circle"]

--- types ---
twice :: Describe a => a -> [String]
main :: Node -> [String]
