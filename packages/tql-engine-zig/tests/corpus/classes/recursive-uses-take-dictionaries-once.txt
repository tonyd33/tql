title: a recursive binding takes its dictionaries once
grammar: typescript
asserts: simplified, values

`member` is generalized over `Eq a`. It takes the dictionary once, outside
a monomorphic copy of itself that its recursive call reaches directly, so
the call passes no dictionary (Static semantics: Bindings).

--- tql ---
member x ys = case ys of {
  Nil -> False;
  Cons y rest -> if x == y then True else member x rest;
};

main = Filter (\_ -> [member 2 [1, 2, 3], member "b" ["a"]]);

--- source ---
class A {}

--- values ---
[true,false]

--- simplified ---
member = \d ->
  letrec
    member = \x -> \ys -> case ys of
      Nil -> False
      Cons y rest -> case d of
        dict[Eq] m -> case m x y of
          False -> member x rest
          True -> True
  in
  member
main = \_ -> Cons (member instance[Eq,Int] 2 (Cons 1 (Cons 2 (Cons 3 Nil)))) (Cons (member instance[Eq,String] "b" (Cons "a" Nil)) Nil)
