import Data.Int (div, mod);
import Data.Maybe (Maybe(..));

half n = if n `mod` 2 == 0 then Just (n `div` 2) else Nothing;

main = Kleisli $ \_ -> [traverse half [2, 4], traverse half [2, 3]];
