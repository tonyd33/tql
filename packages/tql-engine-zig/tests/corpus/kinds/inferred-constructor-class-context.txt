title: an inferred context constrains a constructor variable
grammar: typescript
asserts: types, values

--- tql ---
class Mappable f where {
  mapf :: (a -> b) -> f a -> f b;
};

instance Mappable List where {
  mapf g xs = case xs of {
    [] -> [];
    h : t -> g h : mapf g t;
  };
};

twice g x = mapf g (mapf g x);

main = Filter (\_ -> twice (+ 1) [1, 2]);

class A {}

[3,4]
--- types ---
